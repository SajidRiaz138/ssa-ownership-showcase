import io.fabric8.kubernetes.api.model.ConfigMap;
import io.fabric8.kubernetes.client.KubernetesClient;

import io.fabric8.kubernetes.client.Watcher;
import io.fabric8.kubernetes.client.WatcherException;

/**
 * The runnable application: it is the RUNTIME SOURCE OF TRUTH for the field
 * {@code <configMap>.data.<key>} and enforces it through reconciliation — the "application owns it
 * at runtime, not the chart" resolution to the SSA two-writer problem.
 *
 * <p><b>Note on "ownership".</b> This is operational enforcement, not named server-side-apply
 * field ownership: {@link KubeConfigMapStore} uses ordinary resourceVersion-guarded updates, so
 * the application does not register an SSA field manager in {@code managedFields} for this key.
 * The fix here is organisational (the chart stops writing the key; the app enforces it). Named SSA
 * field-manager behaviour — co-ownership conflicts and an ownership handoff — is demonstrated
 * separately by the SchemaBinding operator scenario, not by this ConfigMap path.
 *
 * <h2>Composition</h2>
 * <ul>
 *   <li>{@link KubeConfigMapStore} — the fabric8 adapter (reads/conditional-writes the field).</li>
 *   <li>{@link Reconciler} — observed-destination enforcement (UPSERT) of the derived value.</li>
 *   <li>{@link ReconcileCoordinator} — runtime coordination: coalesces triggers, retries FAILED
 *       passes, periodic resync, terminal closed state. (The Reconciler compares desired vs
 *       observed and skips redundant writes; that is NOT coalescing — the coordinator provides
 *       coalescing and retry.)</li>
 *   <li>{@link WatchLifecycle} — manages a watch on the SOURCE ConfigMap; source events and each
 *       (re)install trigger a reconcile. Watch failures recover via
 *       {@link ExecutorRecoveryScheduler}.</li>
 * </ul>
 * All reconcile work runs on a single-thread worker (NOT on the watch callback thread); triggers
 * are routed through the coordinator, which is shutdown-aware so a late event cannot throw an
 * executor rejection back into a fabric8 callback.
 */
public final class ConfigReconcilerApp implements AutoCloseable {

    private static final Logger LOG = LoggerFactory.getLogger(ConfigReconcilerApp.class);

    private final KubernetesClient client;
    private final String namespace;
    private final String sourceConfigMap;
    private final String targetConfigMap;
    private final String key;
    private final ReconcileCoordinator coordinator;
    private final WatchLifecycle lifecycle;
    private final ScheduledExecutorService worker;
    private final ScheduledExecutorService recoveryExec;

    private enum LifeState { NEW, STARTED, CLOSED }
    private final java.util.concurrent.atomic.AtomicReference<LifeState> lifeState =
            new java.util.concurrent.atomic.AtomicReference<>(LifeState.NEW);

    public ConfigReconcilerApp(KubernetesClient client, String namespace,
                               String sourceConfigMap, String targetConfigMap, String key) {
        this(client, namespace, sourceConfigMap, targetConfigMap, key,
                Duration.ofSeconds(2), Duration.ofMinutes(5));
    }

    public ConfigReconcilerApp(KubernetesClient client, String namespace,
                               String sourceConfigMap, String targetConfigMap, String key,
                               Duration retryDelay, Duration resyncInterval) {
        this.client = Objects.requireNonNull(client, "client");
        this.namespace = Objects.requireNonNull(namespace, "namespace");
        this.sourceConfigMap = Objects.requireNonNull(sourceConfigMap, "sourceConfigMap");
        this.targetConfigMap = Objects.requireNonNull(targetConfigMap, "targetConfigMap");
        this.key = Objects.requireNonNull(key, "key");

        // Create executors as LOCALS and only publish them (and the composed dependencies) after
        // all construction succeeds, shutting them down if any later step throws — so a failed
        // construction leaks no daemon executors.
        ScheduledExecutorService w = Executors.newSingleThreadScheduledExecutor(daemon("config-reconciler-worker"));
        ScheduledExecutorService rec = Executors.newSingleThreadScheduledExecutor(daemon("config-reconciler-recovery"));
        try {
            ConfigMapStore store = new KubeConfigMapStore(client, namespace);
            Reconciler reconciler = new Reconciler("config-reconciler", targetConfigMap, key,
                    WriteStrategy.UPSERT, store, this::deriveFromSource);
            this.coordinator = new ReconcileCoordinator(reconciler, w, retryDelay, resyncInterval);
            this.lifecycle = new WatchLifecycle("source-watch", this::watchSource,
                    new ExecutorRecoveryScheduler(rec));
        } catch (RuntimeException | Error constructionFailure) {
            w.shutdownNow();
            rec.shutdownNow();
            throw constructionFailure;
        }
        this.worker = w;
        this.recoveryExec = rec;
    }

    private static java.util.concurrent.ThreadFactory daemon(String name) {
        return r -> { Thread t = new Thread(r, name); t.setDaemon(true); return t; };
    }

    private String deriveFromSource() {
        ConfigMap src = client.configMaps().inNamespace(namespace).withName(sourceConfigMap).get();
        String desired = (src != null && src.getData() != null) ? src.getData().get("desired") : null;
        return desired != null ? desired : "derived-default";
    }

    /**
     * Create a watch on the source ConfigMap. If any post-creation setup throws, the newly created
     * watch is CLOSED before rethrowing — otherwise the lifecycle, which never received the handle,
     * could not close it and it would leak. (Reproduced by the shutdown-during-create test.)
     * Both event delivery and (re)install route through the shutdown-aware coordinator.
     */
    private Watch watchSource(Runnable onClosed) {
        return createThenSetup(
                () -> client.configMaps().inNamespace(namespace).withName(sourceConfigMap)
                        .watch(new Watcher<>() {
                            @Override public void eventReceived(Action action, ConfigMap resource) {
                                LOG.info("source event {}; triggering reconcile", action);
                                coordinator.trigger();   // shutdown-aware; never throws back into fabric8
                            }
                            @Override public void onClose(WatcherException cause) {
                                LOG.info("source watch closed: {}", cause == null ? "(graceful)" : cause.getMessage());
                                onClosed.run();          // hand back to the lifecycle (recovery)
                            }
                        }),
                // reconcile-on-(re)install: resync the derived field for anything missed while
                // disconnected (recovery uses this same path).
                coordinator::trigger);
    }

    /**
     * Create a watch, then run post-creation {@code setup}. If setup throws, the newly created
     * watch is CLOSED (suppressing any close error) before rethrowing — otherwise the lifecycle,
     * which never received the handle, could not close it and it would leak. Package-private and
     * static so the leak-safety can be unit-tested with a fake watch and a throwing setup.
     */
    static Watch createThenSetup(java.util.function.Supplier<Watch> create, Runnable setup) {
        Watch watch = create.get();
        try {
            setup.run();
            return watch;
        } catch (RuntimeException | Error setupFailure) {
            try {
                watch.close();
            } catch (RuntimeException closeFailure) {
                setupFailure.addSuppressed(closeFailure);
            }
            throw setupFailure;
        }
    }

    /** Start: begin periodic resync and install the watch (which triggers the initial reconcile).
     *  One-time: a second {@code start()} (or {@code start()} after {@code close()}) is rejected. */
    public void start() {
        if (!lifeState.compareAndSet(LifeState.NEW, LifeState.STARTED)) {
            throw new IllegalStateException("start() is one-time; current state=" + lifeState.get());
        }
        LOG.info("starting; source={} target={}.{}", sourceConfigMap, targetConfigMap, key);
        coordinator.startPeriodicResync();
        lifecycle.registerWatch();
    }

    /**
     * Bounded, coordinated, IDEMPOTENT shutdown. Deactivate the lifecycle (no further installs),
     * close the coordinator (terminal; cancels retry/resync), then shut the executors down and
     * AWAIT their termination within a budget. {@code shutdownNow()} only requests interruption;
     * termination is what we actually wait for. The caller keeps the Kubernetes client open until
     * this returns. A second call is a no-op.
     */
    @Override
    public void close() {
        if (lifeState.getAndSet(LifeState.CLOSED) == LifeState.CLOSED) {
            return;     // idempotent
        }
        lifecycle.deactivate();
        coordinator.close();
        worker.shutdownNow();
        recoveryExec.shutdownNow();
        awaitTermination(worker, "worker");
        awaitTermination(recoveryExec, "recovery");
    }

    private void awaitTermination(ScheduledExecutorService ex, String name) {
        try {
            if (!ex.awaitTermination(10, TimeUnit.SECONDS)) {
                LOG.warn("{} executor did not terminate within the shutdown budget", name);
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    // accessors for tests
    WatchLifecycle lifecycle() { return lifecycle; }
    void triggerNow() { coordinator.trigger(); }
}

package io.ssaownership.reconciler.app;

import io.fabric8.kubernetes.client.KubernetesClient;
import io.fabric8.kubernetes.client.KubernetesClientBuilder;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.Optional;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;

/**
 * Entrypoint. Configuration via environment:
 *   NAMESPACE          (default: current namespace or "default")
 *   SOURCE_CONFIGMAP   (default: "demo-source")
 *   TARGET_CONFIGMAP   (default: "demo-config")
 *   OWNED_KEY          (default: "message")
 *
 * <p>Runs until SIGTERM, then shuts down with a bounded, coordinated sequence. The subtlety: the
 * JVM waits for shutdown HOOKS to finish, not for arbitrary work on the main thread. So the hook
 * must itself drive (or wait for) cleanup; a hook that merely releases a latch can return — and
 * let the JVM exit — before the main thread finishes closing the app and client. Here the hook
 * signals the main thread AND waits, with a bound, for cleanup to complete.
 */
public final class Main {
    private static final Logger LOG = LoggerFactory.getLogger(Main.class);
    private static final long SHUTDOWN_BUDGET_SECONDS = 25;

    public static void main(String[] args) throws InterruptedException {
        String source = env("SOURCE_CONFIGMAP", "demo-source");
        String target = env("TARGET_CONFIGMAP", "demo-config");
        String key = env("OWNED_KEY", "message");

        KubernetesClient client = new KubernetesClientBuilder().build();
        String ns = Optional.ofNullable(env("NAMESPACE", null))
                .orElseGet(() -> Optional.ofNullable(client.getNamespace()).orElse("default"));
        LOG.info("config-reconciler starting in namespace {}", ns);

        ConfigReconcilerApp app = new ConfigReconcilerApp(client, ns, source, target, key);
        CountDownLatch stop = new CountDownLatch(1);        // "please begin shutdown"
        CountDownLatch cleanupDone = new CountDownLatch(1); // "cleanup finished"

        Thread hook = new Thread(() -> {
            stop.countDown();                               // ask main to begin cleanup
            try {
                // wait, with a bound, for the main thread to finish closing app + client, so the
                // JVM does not exit mid-cleanup.
                if (!cleanupDone.await(SHUTDOWN_BUDGET_SECONDS, TimeUnit.SECONDS)) {
                    LOG.warn("shutdown budget elapsed before cleanup completed");
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }, "shutdown");
        Runtime.getRuntime().addShutdownHook(hook);

        try {
            app.start();
            stop.await();                                   // block until SIGTERM (hook) or error
            LOG.info("shutting down");
        } finally {
            // single, idempotent cleanup path shared by normal exit and SIGTERM:
            try {
                app.close();                                // bounded; awaits worker/recovery termination
            } finally {
                client.close();                             // kept open until app work finished
                cleanupDone.countDown();                    // release the hook
            }
        }
    }

    private static String env(String name, String def) {
        String v = System.getenv(name);
        return (v == null || v.isBlank()) ? def : v;
    }
}

package io.ssaownership.reconciler.app;

import io.ssaownership.reconciler.Reconciler;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.time.Duration;
import java.util.Objects;
import java.util.concurrent.RejectedExecutionException;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.ScheduledFuture;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.locks.ReentrantLock;
import java.util.function.Consumer;

/**
 * Runtime coordination for reconcile passes. It sits between event sources (a watch, a periodic
 * timer, a startup trigger) and the policy engine ({@link Reconciler}), providing the properties
 * the engine deliberately does NOT: trigger coalescing, retry of failed passes, periodic resync,
 * and a terminal closed state. {@link Reconciler} itself is unchanged.
 *
 * <h2>Explicit state model</h2>
 * One lock guards a small state machine — chosen over a pile of booleans so retry/close/trigger
 * interactions are easy to reason about:
 * <pre>
 *   IDLE          nothing in flight
 *   RUNNING       a drain is executing (possibly with a queued immediate follow-up: {@code dirty})
 *   RETRY_PENDING a delayed retry is scheduled after a FAILED pass; identified by {@code retryToken}
 *   CLOSED        terminal
 * </pre>
 *
 * <h2>Guarantees</h2>
 * <ul>
 *   <li><b>At most one drain in flight.</b> Triggers during a RUNNING drain set {@code dirty};
 *       exactly one follow-up pass runs afterwards. N triggers collapse into one — real coalescing.</li>
 *   <li><b>Bounded queue.</b> At most one drain task is queued on the executor at a time.</li>
 *   <li><b>Retry after FAILED.</b> A FAILED pass moves to RETRY_PENDING and schedules a delayed
 *       retry carrying a {@code retryToken}; the retry runs ONLY if its token is still current and
 *       the coordinator is open.</li>
 *   <li><b>A trigger expedites a pending retry.</b> A trigger during RETRY_PENDING invalidates the
 *       token (so the old timer is a no-op even if it already fired), cancels the handle, and
 *       starts an immediate drain — the event is not made to wait out the retry delay.</li>
 *   <li><b>Periodic resync.</b> A timer triggers a pass every interval — a fallback for events
 *       missed during a disconnection and for destination-only drift.</li>
 *   <li><b>Closed is terminal and SOUND.</b> {@link #close()} invalidates the retry token and
 *       cancels both the retry and resync handles (outside the lock). A retry that already fired
 *       or was dequeued finds the token stale / state CLOSED and does NOT call the engine.
 *       Correctness comes from the token/closed check, not from cancellation.</li>
 * </ul>
 */
public final class ReconcileCoordinator implements AutoCloseable {

    private static final Logger LOG = LoggerFactory.getLogger(ReconcileCoordinator.class);

    private enum State { IDLE, RUNNING, RETRY_PENDING, CLOSED }

    private final Reconciler reconciler;
    private final ScheduledExecutorService executor;
    private final long retryDelayMs;
    private final long resyncIntervalMs;
    /** Observes an UNEXPECTED executor rejection while open (a liveness failure), distinct from an
     *  expected rejection during shutdown. Never invoked for close-time rejection. */
    private final Consumer<RejectedExecutionException> onUnexpectedRejection;

    private final ReentrantLock lock = new ReentrantLock();
    private State state = State.IDLE;
    private boolean dirty = false;            // a trigger arrived while RUNNING
    private long retryToken = 0;              // identifies the current RETRY_PENDING reservation
    private ScheduledFuture<?> retryHandle;   // the pending retry's handle (guarded)
    private ScheduledFuture<?> resyncHandle;  // periodic resync handle (guarded)

    public ReconcileCoordinator(Reconciler reconciler, ScheduledExecutorService executor,
                                Duration retryDelay, Duration resyncInterval) {
        this(reconciler, executor, retryDelay, resyncInterval, rej ->
                LOG.error("reconcile coordinator: unexpected executor rejection while open "
                        + "(controller is alive but may be inert)", rej));
    }

    public ReconcileCoordinator(Reconciler reconciler, ScheduledExecutorService executor,
                                Duration retryDelay, Duration resyncInterval,
                                Consumer<RejectedExecutionException> onUnexpectedRejection) {
        this.reconciler = Objects.requireNonNull(reconciler, "reconciler");
        this.executor = Objects.requireNonNull(executor, "executor");
        this.onUnexpectedRejection = Objects.requireNonNull(onUnexpectedRejection, "onUnexpectedRejection");
        this.retryDelayMs = normalizeMillis(retryDelay, "retryDelay");
        this.resyncIntervalMs = normalizeMillis(resyncInterval, "resyncInterval");
    }

    /**
     * Normalise a configured duration to a positive millisecond value: reject null/non-positive,
     * map a positive sub-millisecond value to 1ms (never 0 — a 0 period is rejected by
     * {@code scheduleWithFixedDelay}), and turn conversion overflow into a parameter-named
     * {@link IllegalArgumentException}. Accepted precision is milliseconds.
     */
    private static long normalizeMillis(Duration d, String param) {
        Objects.requireNonNull(d, param);
        if (d.isNegative() || d.isZero()) throw new IllegalArgumentException(param + " must be positive");
        final long ms;
        try { ms = d.toMillis(); }
        catch (ArithmeticException overflow) {
            throw new IllegalArgumentException(param + " too large to represent in milliseconds: " + d, overflow);
        }
        return Math.max(1, ms);     // positive sub-ms -> 1ms
    }

    /** Start periodic resync. Call once after construction. */
    public void startPeriodicResync() {
        lock.lock();
        try {
            if (state == State.CLOSED || resyncHandle != null) return;
            resyncHandle = executor.scheduleWithFixedDelay(this::trigger,
                    resyncIntervalMs, resyncIntervalMs, TimeUnit.MILLISECONDS);
        } catch (RejectedExecutionException rej) {
            handleRejection(rej);
        } finally {
            lock.unlock();
        }
    }

    /**
     * Request a reconcile. Shutdown-aware: a no-op once closed; never throws
     * {@link RejectedExecutionException} back to the caller (e.g. a fabric8 watch callback).
     * <ul>
     *   <li>IDLE → start a drain.</li>
     *   <li>RUNNING → mark {@code dirty}; exactly one follow-up pass will run.</li>
     *   <li>RETRY_PENDING → EXPEDITE: invalidate the retry token, cancel its handle (outside the
     *       lock), and start an immediate drain instead of waiting out the delay.</li>
     * </ul>
     */
    public void trigger() {
        ScheduledFuture<?> toCancel = null;
        boolean submit = false;
        lock.lock();
        try {
            switch (state) {
                case CLOSED -> { return; }
                case RUNNING -> { dirty = true; return; }
                case RETRY_PENDING -> {
                    retryToken++;                 // invalidate the pending retry
                    toCancel = retryHandle;
                    retryHandle = null;
                    state = State.RUNNING;
                    submit = true;
                }
                case IDLE -> { state = State.RUNNING; submit = true; }
            }
        } finally {
            lock.unlock();
        }
        cancelOutsideLock(toCancel);              // expedite: drop the delayed retry
        if (submit) submitDrain();
    }

    private void submitDrain() {
        try {
            executor.execute(this::drain);
        } catch (RejectedExecutionException rej) {
            lock.lock();
            try {
                if (state != State.CLOSED) { state = State.IDLE; handleRejection(rej); }
            } finally {
                lock.unlock();
            }
        }
    }

    private void drain() {
        // Guard BEFORE touching the engine: a dequeued drain/retry must not reconcile after close.
        lock.lock();
        try {
            if (state == State.CLOSED) return;
        } finally {
            lock.unlock();
        }

        Reconciler.Outcome outcome;
        try {
            outcome = reconciler.reconcile();
        } catch (RuntimeException e) {
            outcome = Reconciler.Outcome.FAILED;
            LOG.error("reconcile pass threw unexpectedly", e);
        }

        boolean rerunNow = false;
        boolean scheduleRetry = false;
        long tokenForRetry = 0;
        lock.lock();
        try {
            if (state == State.CLOSED) return;
            if (dirty) {
                dirty = false;
                rerunNow = true;                  // coalesced follow-up; stay RUNNING
            } else if (outcome == Reconciler.Outcome.FAILED) {
                state = State.RETRY_PENDING;
                tokenForRetry = ++retryToken;
                scheduleRetry = true;
            } else {
                state = State.IDLE;               // settled
            }
        } finally {
            lock.unlock();
        }

        if (rerunNow) {
            submitDrain();
        } else if (scheduleRetry) {
            scheduleRetry(tokenForRetry);
        }
    }

    private void scheduleRetry(long token) {
        final ScheduledFuture<?> handle;
        try {
            handle = executor.schedule(() -> runRetry(token), retryDelayMs, TimeUnit.MILLISECONDS);
        } catch (RejectedExecutionException rej) {
            lock.lock();
            try {
                if (state == State.RETRY_PENDING && retryToken == token) {
                    state = State.IDLE;
                    handleRejection(rej);
                }
            } finally {
                lock.unlock();
            }
            return;
        }
        boolean stale;
        lock.lock();
        try {
            // A trigger/close may have superseded this reservation while we were scheduling.
            stale = (state != State.RETRY_PENDING) || (retryToken != token);
            if (!stale) retryHandle = handle;
        } finally {
            lock.unlock();
        }
        if (stale) cancelOutsideLock(handle);
    }

    /** The delayed retry body. Runs only if its token is still current AND not closed. */
    private void runRetry(long token) {
        lock.lock();
        try {
            if (state != State.RETRY_PENDING || retryToken != token) return;  // superseded/closed
            state = State.RUNNING;
            retryHandle = null;
        } finally {
            lock.unlock();
        }
        drain();
    }

    /** Must be called while holding the lock (open state only). */
    private void handleRejection(RejectedExecutionException rej) {
        onUnexpectedRejection.accept(rej);
    }

    @Override
    public void close() {
        ScheduledFuture<?> retry;
        ScheduledFuture<?> resync;
        lock.lock();
        try {
            state = State.CLOSED;
            retryToken++;                         // invalidate any in-flight retry reservation
            retry = retryHandle; retryHandle = null;
            resync = resyncHandle; resyncHandle = null;
        } finally {
            lock.unlock();
        }
        cancelOutsideLock(retry);                 // Invariant: cancel OUTSIDE the lock
        cancelOutsideLock(resync);
    }

    private void cancelOutsideLock(ScheduledFuture<?> f) {
        if (f != null) {
            try { f.cancel(false); } catch (RuntimeException e) { LOG.warn("cancel error (ignored)", e); }
        }
    }

    // ---- test hooks (package-private) ----
    boolean debugRunning()      { lock.lock(); try { return state == State.RUNNING; } finally { lock.unlock(); } }
    boolean debugRetryPending() { lock.lock(); try { return state == State.RETRY_PENDING; } finally { lock.unlock(); } }
    boolean debugDirty()        { lock.lock(); try { return dirty; } finally { lock.unlock(); } }
    boolean debugClosed()       { lock.lock(); try { return state == State.CLOSED; } finally { lock.unlock(); } }
    long debugRetryToken()      { lock.lock(); try { return retryToken; } finally { lock.unlock(); } }
    /** Invoke the retry body with a given token exactly as the scheduled timer would, so a test
     *  can deterministically run a STALE retry (e.g. after close) and prove it does not reconcile. */
    void debugRunRetry(long token) { runRetry(token); }
}

package io.ssaownership.reconciler.k8s;

import io.fabric8.kubernetes.api.model.ConfigMap;
import io.fabric8.kubernetes.client.KubernetesClient;
import io.fabric8.kubernetes.client.KubernetesClientException;
import io.ssaownership.reconciler.ConfigMapStore;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;

/**
 * fabric8-backed {@link ConfigMapStore}, bound to one namespace. Implements the contract exactly,
 * in particular the concurrency-safe {@link #putIfAbsent}: it reads, checks the key, and writes
 * at the OBSERVED resourceVersion; on a 409 conflict it re-reads and re-evaluates (the key may
 * now be present, in which case it preserves it). A naive read-then-write is deliberately avoided.
 *
 * <p><b>UPSERT does not retry conflicts internally.</b> {@link #upsert} performs a single
 * optimistic update at the observed resourceVersion and propagates a 409 conflict as a
 * {@link StoreException}. Retrying a conflicted enforcement is the application's job: the
 * reconcile coordinator retries the pass, which RE-OBSERVES current state before deciding to
 * write again. Keeping the retry at the reconcile level (not inside the adapter) ensures the
 * desired-vs-observed comparison is re-evaluated against fresh state on every attempt.
 *
 * <p><b>Whole-resource update, not key-level SSA.</b> Writes update a copied ConfigMap resource
 * with resourceVersion protection against lost updates; they do NOT establish an independent
 * server-side-apply field manager for the key. Named SSA field ownership is demonstrated by the
 * SchemaBinding operator scenario, not by this adapter.
 */
public final class KubeConfigMapStore implements ConfigMapStore {

    private static final Logger LOG = LoggerFactory.getLogger(KubeConfigMapStore.class);

    private final KubernetesClient client;
    private final String namespace;
    private final int maxConflictRetries;
    /** TEST SEAM: runs inside putIfAbsent AFTER the read/absent-check, BEFORE the write. No-op in
     *  production; a test injects a concurrent insert here to exercise the conflict path. */
    private final Runnable beforeConditionalWrite;

    public KubeConfigMapStore(KubernetesClient client, String namespace) {
        this(client, namespace, 5);
    }

    public KubeConfigMapStore(KubernetesClient client, String namespace, int maxConflictRetries) {
        this(client, namespace, maxConflictRetries, () -> {});
    }

    /** Package-private: adds the pre-write test seam. */
    KubeConfigMapStore(KubernetesClient client, String namespace, int maxConflictRetries,
                       Runnable beforeConditionalWrite) {
        this.client = Objects.requireNonNull(client, "client");
        this.namespace = Objects.requireNonNull(namespace, "namespace");
        if (maxConflictRetries < 0) throw new IllegalArgumentException("maxConflictRetries < 0");
        this.maxConflictRetries = maxConflictRetries;
        this.beforeConditionalWrite = Objects.requireNonNull(beforeConditionalWrite, "beforeConditionalWrite");
    }

    @Override
    public Optional<String> get(String configMap, String key) {
        final ConfigMap cm;
        try {
            cm = client.configMaps().inNamespace(namespace).withName(configMap).get();
        } catch (KubernetesClientException e) {
            throw new StoreException("get " + configMap + " failed", e);   // NOT mapped to empty
        }
        if (cm == null || cm.getData() == null) {
            return Optional.empty();            // ConfigMap or key absent
        }
        return Optional.ofNullable(cm.getData().get(key));   // present "" -> Optional.of("")
    }

    @Override
    public void upsert(String configMap, String key, String value) {
        Objects.requireNonNull(value, "value");
        try {
            ConfigMap cm = require(configMap);  // must exist (do not create)
            Map<String, String> data = cm.getData() == null ? new HashMap<>() : new HashMap<>(cm.getData());
            data.put(key, value);
            cm.setData(data);
            // replace at the observed resourceVersion (cm carries it) -> optimistic concurrency
            client.configMaps().inNamespace(namespace).resource(cm).update();
        } catch (KubernetesClientException e) {
            throw new StoreException("upsert " + configMap + "." + key + " failed", e);
        }
    }

    @Override
    public boolean putIfAbsent(String configMap, String key, String value) {
        Objects.requireNonNull(value, "value");
        KubernetesClientException last = null;
        for (int attempt = 0; attempt <= maxConflictRetries; attempt++) {
            ConfigMap cm;
            try {
                cm = require(configMap);
            } catch (KubernetesClientException e) {
                throw new StoreException("putIfAbsent read " + configMap + " failed", e);
            }
            Map<String, String> data = cm.getData();
            if (data != null && data.containsKey(key)) {
                return false;                   // present -> preserve, done
            }
            Map<String, String> next = data == null ? new HashMap<>() : new HashMap<>(data);
            next.put(key, value);
            cm.setData(next);
            if (attempt == 0) {
                beforeConditionalWrite.run();   // TEST SEAM: inject a concurrent insert here
            }
            try {
                client.configMaps().inNamespace(namespace).resource(cm).update();  // at observed rv
                return true;
            } catch (KubernetesClientException e) {
                if (e.getCode() == 409) {       // conflict: someone else wrote; re-read & re-evaluate
                    last = e;
                    LOG.info("putIfAbsent {}.{} conflict (attempt {}); re-reading", configMap, key, attempt);
                    continue;
                }
                throw new StoreException("putIfAbsent " + configMap + "." + key + " failed", e);
            }
        }
        throw new StoreException("putIfAbsent " + configMap + "." + key
                + " exhausted conflict retries", last);
    }

    private ConfigMap require(String configMap) {
        ConfigMap cm = client.configMaps().inNamespace(namespace).withName(configMap).get();
        if (cm == null) {
            throw new StoreException("ConfigMap " + configMap + " does not exist in " + namespace);
        }
        return cm;
    }
}

import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicReference;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Deterministic unit tests for {@link ReconcileCoordinator}. A real {@link Reconciler} is driven
 * over a controllable in-test {@link ConfigMapStore}; a real single-thread scheduled executor
 * serialises drains. Latches signal pass ENTRY and SETTLEMENT — no spin loops, no sleeps as
 * correctness evidence.
 */
class ReconcileCoordinatorTest {

    /** Controllable store: signals each pass entry, can block a pass, can force FAILED. */
    static final class ControllableStore implements ConfigMapStore {
        final AtomicInteger passes = new AtomicInteger();
        volatile CountDownLatch entered;          // counted down at each get() entry (if set)
        volatile CountDownLatch blockOnPass;      // if set, the next get() blocks on it
        volatile boolean failNext = false;
        final AtomicReference<String> value = new AtomicReference<>(null);

        @Override public Optional<String> get(String cm, String key) {
            passes.incrementAndGet();
            CountDownLatch e = entered; if (e != null) e.countDown();
            CountDownLatch gate = blockOnPass;
            if (gate != null) {
                blockOnPass = null;
                try { if (!gate.await(5, TimeUnit.SECONDS)) throw new AssertionError("gate"); }
                catch (InterruptedException ie) { Thread.currentThread().interrupt(); throw new AssertionError(ie); }
            }
            if (failNext) { failNext = false; throw new StoreException("injected"); }
            return Optional.ofNullable(value.get());
        }
        @Override public void upsert(String cm, String key, String v) { value.set(v); }
        @Override public boolean putIfAbsent(String cm, String key, String v) {
            if (value.get() == null) { value.set(v); return true; } return false;
        }
    }

    private ScheduledExecutorService exec;
    private ControllableStore store;
    private ReconcileCoordinator coord;

    private ReconcileCoordinator build(Duration retry, Duration resync) {
        exec = Executors.newSingleThreadScheduledExecutor();
        store = new ControllableStore();
        Reconciler r = new Reconciler("t", "cm", "k", WriteStrategy.UPSERT, store, () -> "desired");
        coord = new ReconcileCoordinator(r, exec, retry, resync);
        return coord;
    }

    @AfterEach
    void tearDown() throws Exception {
        if (coord != null) coord.close();
        if (exec != null) {
            exec.shutdownNow();
            assertTrue(exec.awaitTermination(5, TimeUnit.SECONDS), "executor must terminate");
        }
    }

    @Test @Timeout(10)
    void triggers_during_a_pass_coalesce_into_exactly_one_follow_up() throws Exception {
        build(Duration.ofSeconds(1), Duration.ofMinutes(10));
        CountDownLatch entered = new CountDownLatch(1);
        CountDownLatch release = new CountDownLatch(1);
        store.entered = entered;
        store.blockOnPass = release;                       // pass 1 blocks inside get()

        coord.trigger();                                   // starts drain; pass 1 blocks
        assertTrue(entered.await(5, TimeUnit.SECONDS), "pass 1 must enter");
        assertTrue(coord.debugRunning());

        for (int i = 0; i < 50; i++) coord.trigger();      // many triggers while blocked
        assertTrue(coord.debugDirty(), "triggers during a pass set the dirty flag");

        release.countDown();                               // let pass 1 finish; one follow-up runs
        awaitSettled();
        assertEquals(2, store.passes.get(), "50 triggers during a pass collapse into exactly one follow-up");
        assertFalse(coord.debugDirty());
    }

    @Test @Timeout(10)
    void failed_outcome_schedules_a_retry_without_a_new_trigger() throws Exception {
        build(Duration.ofMillis(150), Duration.ofMinutes(10));
        store.value.set("desired");
        store.failNext = true;                             // first pass FAILED
        coord.trigger();
        awaitPasses(2);                                    // the retry (no new trigger) runs
        assertTrue(store.passes.get() >= 2, "FAILED must schedule a retry pass without a new trigger");
    }

    @Test @Timeout(10)
    void close_cancels_a_pending_retry_and_prevents_post_close_reconcile() throws Exception {
        // The BLOCKING correctness test: fail -> retry reserved -> close -> run the (canceled)
        // retry body DELIBERATELY -> assert NO additional reconcile pass. Correctness must come
        // from the token/closed check, not from cancellation.
        build(Duration.ofHours(1), Duration.ofMinutes(10));   // long retry so it never fires on its own
        store.failNext = true;
        CountDownLatch firstEntered = new CountDownLatch(1);
        store.entered = firstEntered;
        coord.trigger();
        assertTrue(firstEntered.await(5, TimeUnit.SECONDS), "first pass ran");
        // wait until RETRY_PENDING is established
        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
        while (!coord.debugRetryPending() && System.nanoTime() < deadline) { /* short */ Thread.yield(); }
        assertTrue(coord.debugRetryPending(), "a retry must be pending after FAILED");
        long tokenBefore = coord.debugRetryToken();
        int passesBefore = store.passes.get();

        coord.close();
        assertTrue(coord.debugClosed());
        assertNotEquals(tokenBefore, coord.debugRetryToken(), "close must invalidate the retry token");

        // Deterministically run the retry body with its ORIGINAL (now stale) token, exactly as a
        // dequeued timer would. The token/closed check must make it a no-op — NOT reconcile.
        coord.debugRunRetry(tokenBefore);
        // And a direct trigger after close must also be a no-op:
        coord.trigger();
        assertEquals(passesBefore, store.passes.get(),
                "no reconcile pass may run after close (stale retry and trigger both declined)");
    }

    @Test @Timeout(10)
    void a_trigger_expedites_a_pending_retry() throws Exception {
        // After FAILED, a real trigger must EXPEDITE: cancel the long delayed retry and run an
        // immediate pass, not wait out the retry delay.
        build(Duration.ofHours(1), Duration.ofMinutes(10));   // retry delay effectively "never"
        store.failNext = true;
        CountDownLatch firstEntered = new CountDownLatch(1);
        store.entered = firstEntered;
        coord.trigger();
        assertTrue(firstEntered.await(5, TimeUnit.SECONDS), "first (failing) pass ran");
        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
        while (!coord.debugRetryPending() && System.nanoTime() < deadline) Thread.yield();
        assertTrue(coord.debugRetryPending(), "retry pending after FAILED");

        CountDownLatch secondEntered = new CountDownLatch(1);
        store.entered = secondEntered;
        coord.trigger();                                   // must expedite, not wait the 1h delay
        assertTrue(secondEntered.await(5, TimeUnit.SECONDS),
                "a trigger must expedite the pending retry into an immediate pass");
        awaitSettled();
        assertEquals(2, store.passes.get(), "exactly the failing pass + the expedited pass (no extra)");
    }

    @Test @Timeout(10)
    void closed_coordinator_declines_new_work() {
        build(Duration.ofSeconds(1), Duration.ofMinutes(10));
        coord.close();
        assertTrue(coord.debugClosed());
        coord.trigger();
        assertFalse(coord.debugRunning(), "closed coordinator starts no drain");
        assertEquals(0, store.passes.get(), "no pass after close");
    }

    @Test @Timeout(10)
    void trigger_after_executor_shutdown_does_not_throw() {
        build(Duration.ofSeconds(1), Duration.ofMinutes(10));
        exec.shutdownNow();
        assertDoesNotThrow(coord::trigger,
                "a trigger must swallow executor rejection (never throw back into a watch callback)");
    }

    // ---- helpers ----
    private void awaitSettled() throws InterruptedException {
        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
        while (coord.debugRunning() && System.nanoTime() < deadline) Thread.sleep(5);
        assertFalse(coord.debugRunning(), "coordinator must settle");
    }
    private void awaitPasses(int n) throws InterruptedException {
        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
        while (store.passes.get() < n && System.nanoTime() < deadline) Thread.sleep(5);
    }
}

package io.ssaownership.reconciler;

import java.util.Optional;

/**
 * Narrow abstraction over the one capability the reconcile engine needs: read, and conditionally
 * write, a single key in a target ConfigMap. Keeping it narrow makes {@link Reconciler} unit
 * testable with an in-memory fake, and confines the fabric8 client to one adapter.
 *
 * <h2>Contract (an implementation MUST honour all of these)</h2>
 * <ul>
 *   <li><b>Namespace.</b> An instance is bound to exactly one namespace, fixed at construction.
 *       {@code configMap} arguments are names within that namespace.</li>
 *   <li><b>Missing ConfigMap or key -> empty read.</b> {@link #get} returns {@link Optional#empty()}
 *       when EITHER the ConfigMap does not exist OR it exists but the key is absent (these two
 *       "absent" cases are not distinguished by {@code get}). A present key whose value is the
 *       empty string returns {@code Optional.of("")}. Writes DO distinguish them:
 *       {@link #upsert}/{@link #putIfAbsent} MUST fail with {@link StoreException} if the ConfigMap
 *       object does not exist (they do NOT create it — that is the chart's responsibility).</li>
 *   <li><b>Read errors are not absence.</b> Any API failure during {@link #get} (auth, network,
 *       server error) MUST throw {@link StoreException}, never be mapped to {@link Optional#empty()}.
 *       Empty means "ConfigMap or key absent" — never "the read failed".</li>
 *   <li><b>putIfAbsent is concurrency-safe at the API boundary.</b> The existence check and the
 *       write MUST be atomic with respect to other writers (optimistic concurrency on the
 *       observed resourceVersion, re-reading and re-evaluating on conflict). A naive
 *       read-then-write that can clobber a concurrent insert does NOT satisfy this contract.</li>
 *   <li><b>Preserve unrelated data and user-managed metadata.</b> Writes touch only {@code key};
 *       they MUST preserve other data keys and user-managed labels/annotations. They do NOT
 *       prevent the API server maintaining system metadata (e.g. {@code resourceVersion},
 *       managed-fields) — that normal server maintenance is expected.</li>
 *   <li><b>Retry exhaustion.</b> If conflict retries are exhausted, the operation MUST throw
 *       {@link StoreException} rather than return a misleading success/failure.</li>
 *   <li><b>Reject null.</b> Null values are rejected ({@link NullPointerException}); null is not
 *       a representable ConfigMap value.</li>
 * </ul>
 *
 * <p>Note on write policy: {@link #upsert} is a <em>write policy</em> ("create or replace the
 * key"), not a licence to force Kubernetes server-side-apply field-ownership conflicts. SSA
 * ownership is a separate concern handled by the apply mechanism, not by this interface.
 */
public interface ConfigMapStore {

    /** Thrown for any store failure (API error, missing ConfigMap on write, retry exhaustion). */
    class StoreException extends RuntimeException {
        public StoreException(String message, Throwable cause) { super(message, cause); }
        public StoreException(String message) { super(message); }
    }

    /**
     * @return the current value of {@code key} in {@code configMap}; empty if the ConfigMap does
     *         not exist OR the key is absent (these two "absent" cases are not distinguished).
     *         A present empty-string value returns {@code Optional.of("")}. Throws
     *         {@link StoreException} on API failure (never maps error to empty).
     */
    Optional<String> get(String configMap, String key);

    /**
     * Create or replace {@code key = value}, preserving unrelated content.
     * @throws StoreException if the ConfigMap does not exist or the write ultimately fails.
     * @throws NullPointerException if value is null.
     */
    void upsert(String configMap, String key, String value);

    /**
     * Set {@code key = value} only if the key is currently absent, atomically with respect to
     * concurrent writers (re-read and re-evaluate on conflict). Preserves an existing value.
     * @return true if it wrote (key was absent), false if an existing value was left untouched.
     * @throws StoreException if the ConfigMap does not exist or the write ultimately fails.
     * @throws NullPointerException if value is null.
     */
    boolean putIfAbsent(String configMap, String key, String value);
}

package io.ssaownership.reconciler;

import java.time.Duration;
import java.util.Objects;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.ScheduledFuture;
import java.util.concurrent.TimeUnit;

/**
 * A {@link RecoveryScheduler} backed by a {@link ScheduledExecutorService}.
 *
 * <h2>Executor ownership (deliberate)</h2>
 * This adapter does NOT own the executor it is given and never shuts it down — several
 * {@link WatchLifecycle} instances may share one executor, so one lifecycle deactivating must not
 * stop the others' recovery. If you want a dedicated, closeable scheduler, use
 * {@link #createOwning(String)} which creates (and, via {@link #close()}, shuts down) its own
 * single-thread executor.
 *
 * <h2>Contract honoured</h2>
 * <ul>
 *   <li>Tasks run on an executor thread, never inline on the scheduling caller's stack.</li>
 *   <li>{@link #schedule} returns a per-task {@link Cancellable} that cancels only that task.</li>
 *   <li>A rejected submission (executor shut down, queue bounded) throws
 *       {@link java.util.concurrent.RejectedExecutionException}, which {@link WatchLifecycle}
 *       translates into its DEGRADED failure path.</li>
 *   <li>Invalid (negative) delays are rejected with {@link IllegalArgumentException}.</li>
 * </ul>
 */
public final class ExecutorRecoveryScheduler implements RecoveryScheduler, AutoCloseable {

    private final ScheduledExecutorService executor;
    private final boolean owned;

    /** Use a shared, externally-owned executor; {@link #close()} will NOT shut it down. */
    public ExecutorRecoveryScheduler(ScheduledExecutorService executor) {
        this(executor, false);
    }

    private ExecutorRecoveryScheduler(ScheduledExecutorService executor, boolean owned) {
        this.executor = Objects.requireNonNull(executor, "executor");
        this.owned = owned;
    }

    /** Create a scheduler that OWNS a dedicated single-thread executor, shut down by {@link #close()}. */
    public static ExecutorRecoveryScheduler createOwning(String threadName) {
        ScheduledExecutorService ex = java.util.concurrent.Executors.newSingleThreadScheduledExecutor(r -> {
            Thread t = new Thread(r, threadName);
            t.setDaemon(true);
            return t;
        });
        return new ExecutorRecoveryScheduler(ex, true);
    }

    @Override
    public Cancellable schedule(Duration delay, Runnable task) {
        Objects.requireNonNull(delay, "delay");
        Objects.requireNonNull(task, "task");
        if (delay.isNegative()) {
            throw new IllegalArgumentException("delay must not be negative: " + delay);
        }
        // Accepts nonnegative durations whose toMillis() conversion fits in a long. A positive
        // sub-millisecond duration is submitted as 1ms (never silently 0). For durations above 1ms
        // the conversion truncates fractional milliseconds — it does NOT round every positive
        // duration upward. Actual scheduling precision and extreme-delay handling depend on the
        // backing executor; conversion overflow is rejected as IllegalArgumentException.
        long millis;
        try {
            millis = delay.toMillis();
        } catch (ArithmeticException overflow) {
            throw new IllegalArgumentException("delay too large to represent in millis: " + delay, overflow);
        }
        if (millis == 0 && !delay.isZero()) {
            millis = 1;                       // positive sub-ms -> minimum representable, not dropped
        }
        // May throw RejectedExecutionException — intentionally propagated to the caller.
        ScheduledFuture<?> future = executor.schedule(task, millis, TimeUnit.MILLISECONDS);
        return () -> future.cancel(false);   // best-effort; a running task is not interrupted
    }

    /** Shuts down the executor ONLY if this scheduler created it. No-op for a shared executor. */
    @Override
    public void close() {
        if (owned) {
            executor.shutdownNow();
        }
    }
}

package io.ssaownership.reconciler;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.Objects;
import java.util.Optional;
import java.util.function.Supplier;

/**
 * The reconcile engine: derive the desired value for a single owned field, then bring the target
 * ConfigMap to that value using the configured {@link WriteStrategy}. The application — not the
 * Helm chart — writes this field at runtime; this engine is the application's writer.
 *
 * <h2>Contract (observed-destination enforcement)</h2>
 * Each pass compares the desired value against the <em>observed destination</em>, not a cache of
 * what was last derived. Therefore a pass repairs destination drift:
 * <ul>
 *   <li>{@link WriteStrategy#UPSERT}: read the key; if it already equals desired, skip; otherwise
 *       create-or-replace it. This repairs drift <em>when a pass runs</em>.</li>
 *   <li>{@link WriteStrategy#PUT_IF_ABSENT}: call the store's concurrency-safe conditional write;
 *       an existing value is preserved (never reverted).</li>
 * </ul>
 *
 * <h2>What this does NOT claim</h2>
 * <ul>
 *   <li>It is NOT continuous enforcement. It "repairs drift when reconciliation runs." WHEN a
 *       pass runs is decided by event wiring (destination watch and/or periodic resync), which
 *       lives outside this class. Between passes, destination drift can persist.</li>
 *   <li>It does NOT establish Kubernetes SSA field ownership. Write policy is not field-manager
 *       ownership; that is a property of the apply mechanism, demonstrated elsewhere in the repo.</li>
 *   <li>Comparing desired to observed avoids unnecessary writes; it is NOT a transaction. Another
 *       actor may change the destination immediately after a write. Eventual correction comes
 *       from the next pass, not from this method.</li>
 * </ul>
 */
public final class Reconciler {

    private static final Logger LOG = LoggerFactory.getLogger(Reconciler.class);

    private final String name;
    private final String targetConfigMap;
    private final String key;
    private final WriteStrategy strategy;
    private final ConfigMapStore store;
    /** Derives the desired value from current inputs. */
    private final Supplier<String> deriveDesired;

    public Reconciler(String name,
                      String targetConfigMap,
                      String key,
                      WriteStrategy strategy,
                      ConfigMapStore store,
                      Supplier<String> deriveDesired) {
        this.name = Objects.requireNonNull(name, "name");
        this.targetConfigMap = Objects.requireNonNull(targetConfigMap, "targetConfigMap");
        this.key = Objects.requireNonNull(key, "key");
        this.strategy = Objects.requireNonNull(strategy, "strategy");
        this.store = Objects.requireNonNull(store, "store");
        this.deriveDesired = Objects.requireNonNull(deriveDesired, "deriveDesired");
    }

    /** Outcome of one reconcile pass — explicit so callers and tests can assert behaviour. */
    public enum Outcome {
        /** A write was performed. */                         WROTE,
        /** UPSERT: destination already equalled desired. */  SKIPPED_IN_SYNC,
        /** PUT_IF_ABSENT: key already present, preserved. */ SKIPPED_PRESENT,
        /** Derivation or a store call failed; retryable. */  FAILED
    }

    /**
     * Run one reconcile pass. Converts derivation and store {@link RuntimeException}s into
     * {@link Outcome#FAILED} (it does NOT catch {@link Error}).
     *
     * <p><b>Failure semantics (precise).</b> On {@link Outcome#FAILED} no local success state is
     * advanced, so a subsequent pass is permitted. This does NOT prove the destination is
     * unchanged: a store write may have committed remotely before the client observed a timeout,
     * so a failed call has an <em>unknown remote outcome</em>. The next pass re-observes the
     * destination (UPSERT) or conditionally writes it (PUT_IF_ABSENT) rather than assuming either
     * result. "Retryable" means another pass is permitted — not that the failure is transient;
     * invalid input, authorization errors, or a missing ConfigMap will keep failing until an
     * operator intervenes.
     *
     * <p>{@code synchronized} serialises passes of THIS instance only. It provides no
     * coordination across instances or pods. Callers should run passes on worker execution
     * (not directly on a watch callback thread) and coalesce repeated triggers.
     */
    public synchronized Outcome reconcile() {
        final String desired;
        try {
            desired = Objects.requireNonNull(deriveDesired.get(), "derived value must not be null");
        } catch (RuntimeException e) {
            LOG.error("[{}] could not derive desired value; no store write attempted for {}.{}",
                    name, targetConfigMap, key, e);
            return Outcome.FAILED;
        }

        try {
            return switch (strategy) {
                case UPSERT -> upsertEnforce(desired);
                case PUT_IF_ABSENT -> seedIfAbsent(desired);
            };
        } catch (RuntimeException e) {   // e.g. ConfigMapStore.StoreException
            LOG.error("[{}] reconciliation of {}.{} failed (remote outcome unknown); a subsequent pass re-evaluates the configured policy",
                    name, targetConfigMap, key, e);
            return Outcome.FAILED;
        }
    }

    /** UPSERT: compare desired to OBSERVED destination; write only if they differ. */
    private Outcome upsertEnforce(String desired) {
        Optional<String> observed = store.get(targetConfigMap, key);   // may throw -> FAILED
        if (observed.filter(desired::equals).isPresent()) {
            LOG.debug("[{}] {}.{} already in sync; no write", name, targetConfigMap, key);
            return Outcome.SKIPPED_IN_SYNC;
        }
        store.upsert(targetConfigMap, key, desired);
        LOG.info("[{}] enforced {}.{} (upsert)", name, targetConfigMap, key);
        return Outcome.WROTE;
    }

    /** PUT_IF_ABSENT: concurrency-safe seed; preserve any existing value. No diagnostic read. */
    private Outcome seedIfAbsent(String desired) {
        boolean wrote = store.putIfAbsent(targetConfigMap, key, desired);  // may throw -> FAILED
        if (wrote) {
            LOG.info("[{}] seeded {}.{} (put-if-absent)", name, targetConfigMap, key);
            return Outcome.WROTE;
        }
        // false is a successful policy decision: a value already exists and we preserve it.
        // DEBUG, not INFO: this is the steady-state outcome of a frequently-running reconcile.
        LOG.debug("[{}] {}.{} already present; left untouched", name, targetConfigMap, key);
        return Outcome.SKIPPED_PRESENT;
    }
}

package io.ssaownership.reconciler;

import java.time.Duration;

/**
 * Delayed, cancellable execution — the ONLY thing {@link WatchLifecycle} needs from
 * infrastructure for recovery. Deliberately dumb: it knows nothing about generations, backoff,
 * eligibility, or coalescing. All of that policy lives in the lifecycle; the scheduler only runs
 * a task after a delay and lets the caller cancel that particular task.
 *
 * <h2>Contract</h2>
 * <ul>
 *   <li>{@link #schedule} runs {@code task} after approximately {@code delay}, on an EXECUTOR
 *       thread — never inline on the scheduling caller's stack. (Executor ownership is
 *       implementation-specific; a shared executor remains externally owned.)</li>
 *   <li>The returned {@link Cancellable#cancel()} is <b>best-effort</b>: a task already running
 *       cannot be un-run. The lifecycle's own eligibility check (is this recovery still the
 *       current reservation of an active lifecycle?) is the correctness mechanism; cancellation
 *       is an optimisation.</li>
 *   <li><b>Required backing-executor behaviour:</b> submissions execute asynchronously, and a
 *       submission that cannot be accepted MUST throw
 *       {@link java.util.concurrent.RejectedExecutionException} (or another runtime exception),
 *       so {@link WatchLifecycle} can surface it as DEGRADED. An executor configured with a
 *       SILENTLY-DISCARDING rejection policy violates this contract: the lifecycle would receive
 *       a handle for a task that never runs instead of an observable rejection. Use the default
 *       (throwing) rejection policy.</li>
 *   <li><b>Delay range/precision:</b> delays are honoured at MILLISECOND precision. A positive
 *       sub-millisecond delay is treated as the minimum representable delay (not zero/not
 *       dropped) by well-behaved implementations; a negative delay is rejected; excessively large
 *       delays beyond the implementation's representable range are rejected. Implementations
 *       document their exact supported range.</li>
 * </ul>
 */
public interface RecoveryScheduler {

    Cancellable schedule(Duration delay, Runnable task);

    interface Cancellable {
        /** Best-effort cancel. A task that has already started may still complete. */
        void cancel();
    }
}

package io.ssaownership.reconciler;

import io.fabric8.kubernetes.client.Watch;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.time.Duration;
import java.util.Objects;
import java.util.function.LongSupplier;
import java.util.concurrent.locks.ReentrantLock;

/**
 * Lifecycle for a single Kubernetes {@link Watch}, correct under concurrent registration, late
 * and duplicate close callbacks, close-before-install, shutdown races, and recovery scheduling
 * races. Recovery POLICY lives here; delayed execution is delegated to a {@link RecoveryScheduler}.
 *
 * <h2>The reservation protocol (the heart of the design)</h2>
 * The unit of work is an {@link Attempt}, whose identity is a monotonic generation claimed UNDER
 * the lock. The crucial property: once an attempt is authorized under the lock, that SAME attempt
 * is carried through creation and install — no later step re-mints a new identity after
 * validating an old one. {@code registerWatch()} and {@code runRecovery(reservation)} both reduce
 * to {@link #createAndInstall(Attempt)}:
 * <pre>
 *   createAndInstall(attempt):
 *     factory.create(...)                    // OUTSIDE the lock
 *     under lock: install candidate IFF attempt is still the current authorized attempt
 *                 AND not shut down AND its close notification has not invalidated the attempt;
 *                 else reject
 *     closeQuietly(rejected/superseded)      // OUTSIDE the lock
 * </pre>
 *
 * <h2>Invariants (see lifecycle ADR)</h2>
 * <ol>
 *   <li>Identity survives execution: an authorized attempt is never silently swapped for another.</li>
 *   <li>Only the current eligible attempt installs; superseded candidates are cleaned up.</li>
 *   <li>Explicit registration invalidates+detaches pending recovery under the lock; cancellation
 *       happens outside it.</li>
 *   <li>Closing the old installed watch detaches it; it does NOT invalidate an authorized replacement.</li>
 *   <li>Recovery completion/rejection/cancellation cannot modify a newer reservation's state.</li>
 *   <li>Shutdown is terminal; late factory results / scheduler handles cannot reactivate.</li>
 *   <li>Factory, scheduling, cancellation, and close all happen outside the lock.</li>
 * </ol>
 */
public final class WatchLifecycle {

    private static final Logger LOG = LoggerFactory.getLogger(WatchLifecycle.class);

    public enum Health {
        /** Lifecycle enabled and not degraded. Does NOT mean a functioning watch is currently
         *  installed — the lifecycle is ACTIVE before the first watch exists and during delayed
         *  recovery. Do not use this alone as a readiness signal; a watch may be absent/pending. */
        ACTIVE,
        /** Terminal shutdown. */
        DEACTIVATED,
        /** Recovery scheduling was rejected; needs an external trigger (see ADR-0002). */
        DEGRADED
    }

    /** An authorized unit of creation/installation work, identified by generation (monotonically
     *  increasing within the process lifetime; long overflow is not a practical concern). */
    private static final class Attempt {
        final long generation;
        Attempt(long generation) { this.generation = generation; }
    }

    /** A reserved recovery, identified by generation, awaiting scheduled execution. */
    private static final class RecoveryReservation {
        final long generation;
        RecoveryScheduler.Cancellable handle;   // set after scheduling; guarded by lock
        RecoveryReservation(long generation) { this.generation = generation; }
    }

    private final String name;
    private final WatchFactory factory;
    private final RecoveryScheduler scheduler;
    private final long baseDelayMs;
    private final long maxDelayMs;
    private final long stableResetAfterNanos;
    private final LongSupplier nanoTime;
    /**
     * TEST SYNCHRONISATION SEAM (not a production extension point). Runs OUTSIDE the lifecycle
     * lock on the RECOVERY path only, after a reservation is consumed and its attempt authorized
     * but before creation is dispatched. Production uses a no-op; a test injects a pause here to
     * deterministically interleave an explicit registration. It must not modify internal state.
     */
    private final Runnable afterRecoveryAuthorized;

    private final ReentrantLock lock = new ReentrantLock();
    // ---- all guarded by lock ----
    private Health health = Health.ACTIVE;
    private long generation = 0;                 // last authorized attempt/reservation id
    private Watch installed;                     // the installed watch (may be null)
    private long installedGeneration = -1;
    private long installedAtNanos = 0;
    private int consecutiveFailures = 0;         // saturating; see nextDelayMsLocked
    private RecoveryReservation pendingRecovery; // at most one

    public WatchLifecycle(String name, WatchFactory factory, RecoveryScheduler scheduler) {
        this(name, factory, scheduler, Duration.ofSeconds(1), Duration.ofSeconds(30),
                Duration.ofSeconds(60), System::nanoTime);
    }

    public WatchLifecycle(String name, WatchFactory factory, RecoveryScheduler scheduler,
                          Duration baseDelay, Duration maxDelay, Duration stableResetAfter,
                          LongSupplier nanoTime) {
        this(name, factory, scheduler, baseDelay, maxDelay, stableResetAfter, nanoTime, () -> {});
    }

    /** Package-private: adds the recovery-path test seam. Public callers get the no-op overload. */
    WatchLifecycle(String name, WatchFactory factory, RecoveryScheduler scheduler,
                   Duration baseDelay, Duration maxDelay, Duration stableResetAfter,
                   LongSupplier nanoTime, Runnable afterRecoveryAuthorized) {
        this.name = Objects.requireNonNull(name, "name");
        if (name.isBlank()) throw new IllegalArgumentException("name must not be blank");
        this.factory = Objects.requireNonNull(factory, "factory");
        this.scheduler = Objects.requireNonNull(scheduler, "scheduler");
        Objects.requireNonNull(baseDelay, "baseDelay");
        Objects.requireNonNull(maxDelay, "maxDelay");
        Objects.requireNonNull(stableResetAfter, "stableResetAfter");
        this.nanoTime = Objects.requireNonNull(nanoTime, "nanoTime");
        this.afterRecoveryAuthorized = Objects.requireNonNull(afterRecoveryAuthorized, "afterRecoveryAuthorized");
        // Validate configuration (fail fast): positive base/stability, maxDelay >= baseDelay.
        if (baseDelay.isNegative() || baseDelay.isZero()) throw new IllegalArgumentException("baseDelay must be positive");
        if (stableResetAfter.isNegative() || stableResetAfter.isZero()) throw new IllegalArgumentException("stableResetAfter must be positive");
        if (maxDelay.compareTo(baseDelay) < 0) throw new IllegalArgumentException("maxDelay must be >= baseDelay");
        // Representable ranges: baseDelay/maxDelay must fit in milliseconds and stableResetAfter in
        // nanoseconds (Duration conversion throws ArithmeticException on overflow). Normalise that
        // overflow into a configuration error that names the offending parameter, rather than
        // letting a raw ArithmeticException escape the constructor.
        this.baseDelayMs = Math.max(1, toMillisOrReject(baseDelay, "baseDelay"));
        this.maxDelayMs = Math.max(this.baseDelayMs, toMillisOrReject(maxDelay, "maxDelay"));
        this.stableResetAfterNanos = toNanosOrReject(stableResetAfter, "stableResetAfter");
    }

    private static long toMillisOrReject(Duration d, String param) {
        try { return d.toMillis(); }
        catch (ArithmeticException overflow) {
            throw new IllegalArgumentException(param + " too large to represent in milliseconds: " + d, overflow);
        }
    }
    private static long toNanosOrReject(Duration d, String param) {
        try { return d.toNanos(); }
        catch (ArithmeticException overflow) {
            throw new IllegalArgumentException(param + " too large to represent in nanoseconds: " + d, overflow);
        }
    }

    /** Explicit (re)registration. Invalidates any pending recovery and authorizes a replacement. */
    public void registerWatch() {
        Attempt attempt;
        RecoveryReservation detachedRecovery;
        lock.lock();
        try {
            if (health == Health.DEACTIVATED) {
                LOG.info("[{}] deactivated; registration refused", name);
                return;
            }
            // Invariant 3: invalidate+detach pending recovery under the lock (cancel outside).
            detachedRecovery = pendingRecovery;
            pendingRecovery = null;
            attempt = new Attempt(++generation);     // authorize THIS attempt
            if (health == Health.DEGRADED) health = Health.ACTIVE;   // actively trying again
        } finally {
            lock.unlock();
        }
        cancelOutsideLock(detachedRecovery);
        createAndInstall(attempt);
    }

    /** The single creation+install path. The authorized {@code attempt} is carried through. */
    private void createAndInstall(Attempt attempt) {
        // Create OUTSIDE the lock; bind the close callback to THIS attempt's generation.
        final Watch candidate;
        try {
            candidate = Objects.requireNonNull(
                    factory.create(() -> onClosed(attempt.generation)),
                    "watch factory returned null");
        } catch (RuntimeException e) {
            LOG.error("[{}] watch creation failed at gen {}", name, attempt.generation, e);
            onCreateFailed(attempt);
            return;
        }

        Watch toClose = null;
        lock.lock();
        try {
            // Close-before-install is enforced here WITHOUT a flag on Attempt, but the guarantee
            // depends on WHEN the close is OBSERVED (the bound onClosed acquiring this lock), not
            // on when the watch physically terminated:
            //   - If this attempt's close notification was PROCESSED BEFORE this installation check,
            //     its onClosed(attempt.generation) already advanced `generation` (reserving
            //     recovery), so attempt.generation != generation and the candidate is rejected here.
            //   - If closure is processed AFTER installation, the candidate is briefly installed and
            //     onClosed then detaches it and arranges recovery (the normal installed-close path).
            // A real client may physically terminate a watch before its callback runs; that latter
            // ordering is handled, not prevented. This is a probe-free distinction between physical
            // termination and lifecycle-observed termination.
            if (health == Health.DEACTIVATED || attempt.generation != generation) {
                // Invariant 2 & 6: superseded, shut down, or its close notification already
                // invalidated this attempt. Reject this candidate.
                LOG.info("[{}] candidate gen {} rejected (health={}, current={})",
                        name, attempt.generation, health, generation);
                toClose = candidate;
            } else {
                toClose = installed;                 // supersede any previous installed watch
                installed = candidate;
                installedGeneration = attempt.generation;
                installedAtNanos = nanoTime.getAsLong();
                health = Health.ACTIVE;
                LOG.info("[{}] watch installed at gen {}", name, attempt.generation);
            }
        } finally {
            lock.unlock();
        }
        closeQuietly(toClose);                       // Invariant 7
    }

    /** Close callback bound to the attempt created at {@code gen}. */
    private void onClosed(long gen) {
        RecoveryReservation reservation = null;
        long delayMs = baseDelayMs;
        lock.lock();
        try {
            if (health == Health.DEACTIVATED) return;        // shutting down

            boolean wasInstalled = (gen == installedGeneration);
            boolean isCurrent = (gen == generation);
            if (!wasInstalled && !isCurrent) {
                // Invariant 4/5: a stale/old watch closed. Detach nothing current, schedule
                // nothing. (If it was the installed one, wasInstalled handles detach below.)
                LOG.info("[{}] ignoring close of non-current watch gen {} (current {})",
                        name, gen, generation);
                return;
            }
            if (wasInstalled) {
                // Backoff decision from ACTUAL live duration (install -> now), not recovery latency.
                long live = nanoTime.getAsLong() - installedAtNanos;
                if (live >= stableResetAfterNanos) consecutiveFailures = 0;
                installed = null;
                installedGeneration = -1;
            }
            // Only arrange recovery if THIS close concerns the current generation. If a newer
            // attempt has already superseded (isCurrent false but wasInstalled true for an old
            // installed watch whose replacement is in flight), we detached the old watch above
            // but must NOT supersede the in-flight replacement (Invariant 4).
            if (!isCurrent) {
                LOG.info("[{}] old installed watch gen {} closed; replacement gen {} in flight, not recovering",
                        name, gen, generation);
                return;
            }
            consecutiveFailures = saturatingInc(consecutiveFailures);
            delayMs = nextDelayMsLocked();
            reservation = new RecoveryReservation(++generation);  // reserve recovery identity
            pendingRecovery = reservation;
        } finally {
            lock.unlock();
        }
        scheduleRecovery(reservation, delayMs);
    }

    private void onCreateFailed(Attempt attempt) {
        RecoveryReservation reservation;
        long delayMs;
        lock.lock();
        try {
            // Only this attempt may arrange recovery, and only if it is still current.
            if (health == Health.DEACTIVATED || attempt.generation != generation) {
                return;      // superseded/shut down: retaining a healthy installed watch if any
            }
            consecutiveFailures = saturatingInc(consecutiveFailures);
            delayMs = nextDelayMsLocked();
            reservation = new RecoveryReservation(++generation);
            pendingRecovery = reservation;
        } finally {
            lock.unlock();
        }
        scheduleRecovery(reservation, delayMs);
    }

    private void scheduleRecovery(RecoveryReservation reservation, long delayMs) {
        final RecoveryScheduler.Cancellable handle;
        try {
            handle = scheduler.schedule(Duration.ofMillis(delayMs), () -> runRecovery(reservation));
        } catch (RuntimeException e) {
            // Rejected scheduling: enter DEGRADED only if this reservation is still current.
            lock.lock();
            try {
                if (pendingRecovery == reservation && health != Health.DEACTIVATED) {
                    pendingRecovery = null;
                    health = Health.DEGRADED;
                    LOG.error("[{}] recovery scheduling rejected at gen {}; DEGRADED (external trigger required)",
                            name, reservation.generation, e);
                }
            } finally {
                lock.unlock();
            }
            return;
        }
        boolean stale;
        lock.lock();
        try {
            // The task may already have run (another thread) or been superseded while scheduling.
            stale = (health == Health.DEACTIVATED) || (pendingRecovery != reservation);
            if (!stale) reservation.handle = handle;   // record only if still current
        } finally {
            lock.unlock();
        }
        if (stale) cancelOutsideLock(handle);          // Invariant 5/7
    }

    /** Scheduler-invoked. Consume the matching reservation and authorize its exact attempt. */
    private void runRecovery(RecoveryReservation reservation) {
        Attempt attempt = null;
        lock.lock();
        try {
            if (health != Health.DEACTIVATED && pendingRecovery == reservation) {
                pendingRecovery = null;
                attempt = new Attempt(reservation.generation);  // SAME identity, now authorized
            } else {
                LOG.info("[{}] recovery gen {} no longer eligible; skipping", name, reservation.generation);
            }
        } finally {
            lock.unlock();
        }
        if (attempt != null) {
            afterRecoveryAuthorized.run();   // TEST SEAM: outside lock, after authorize, before create
            createAndInstall(attempt);       // carries the authorized identity; no re-mint
        }
    }

    /** Terminal shutdown. Invalidates all work; cancels pending recovery; closes installed. */
    public void deactivate() {
        Watch toClose;
        RecoveryReservation toCancel;
        lock.lock();
        try {
            if (health == Health.DEACTIVATED) return;
            health = Health.DEACTIVATED;
            generation++;                              // invalidate in-flight attempts/reservations
            toClose = installed;
            installed = null;
            installedGeneration = -1;
            toCancel = pendingRecovery;
            pendingRecovery = null;
        } finally {
            lock.unlock();
        }
        cancelOutsideLock(toCancel);
        closeQuietly(toClose);
        LOG.info("[{}] deactivated", name);
    }

    public Health health() { lock.lock(); try { return health; } finally { lock.unlock(); } }
    public boolean isActive() { return health() == Health.ACTIVE; }

    // ---- backoff: saturating, validated ------------------------------------------------------
    private static int saturatingInc(int v) { return v == Integer.MAX_VALUE ? v : v + 1; }

    private long nextDelayMsLocked() {
        // Saturating exponential: baseDelayMs * 2^(failures-1), capped at maxDelayMs. No overflow.
        long d = baseDelayMs;
        for (int i = 1; i < consecutiveFailures && d < maxDelayMs; i++) {
            d = (d > maxDelayMs / 2) ? maxDelayMs : d * 2;   // saturate before overflow
        }
        return Math.min(d, maxDelayMs);
    }

    private void closeQuietly(Watch w) {
        if (w == null) return;
        try { w.close(); } catch (RuntimeException e) { LOG.warn("[{}] close error (ignored)", name, e); }
    }
    private void cancelOutsideLock(RecoveryReservation r) {
        if (r != null) cancelOutsideLock(r.handle);
    }
    private void cancelOutsideLock(RecoveryScheduler.Cancellable c) {
        if (c == null) return;
        try { c.cancel(); } catch (RuntimeException e) { LOG.warn("[{}] cancel error (ignored)", name, e); }
    }

    // ---- test hooks (package-private) --------------------------------------------------------
    boolean debugIsInstalled(Watch w) {
        lock.lock(); try { return w != null && w == installed; } finally { lock.unlock(); }
    }
    boolean debugHasPendingRecovery() {
        lock.lock(); try { return pendingRecovery != null; } finally { lock.unlock(); }
    }
    int debugConsecutiveFailures() {
        lock.lock(); try { return consecutiveFailures; } finally { lock.unlock(); }
    }
}

package io.ssaownership.reconciler;

/**
 * How the application writes the field it owns. The two strategies are not interchangeable —
 * choosing the right one is a correctness decision, and getting it wrong is a real bug.
 *
 * <ul>
 *   <li>{@link #UPSERT} — create or replace the target key when reconciliation determines a write
 *       is required (i.e. the observed value differs from desired). Correct when the application
 *       is the source of truth and must enforce its value (e.g. a derived configuration that must
 *       always reflect current inputs).</li>
 *   <li>{@link #PUT_IF_ABSENT} — write only when the key is absent; never replace an existing
 *       value. Correct when the value is a seedable default that another actor may legitimately
 *       have changed, and the app must not revert that change.</li>
 * </ul>
 *
 * <h2>Why the choice is a correctness decision (corrected migration example)</h2>
 * In the migration this models: a derived configuration field used UPSERT; seedable defaults used
 * PUT_IF_ABSENT. Using the wrong one is a real defect:
 * <ul>
 *   <li><b>PUT_IF_ABSENT on derived configuration</b> can <em>preserve stale output</em>: once a
 *       value exists it is never corrected, so the field drifts from its derived truth.</li>
 *   <li><b>UPSERT on editable defaults</b> can <em>revert operator changes</em>: every pass
 *       overwrites a deliberate change back to the default.</li>
 * </ul>
 * Hence the policy is modelled explicitly rather than hard-coded. Note: UPSERT is a write policy,
 * not a licence to force Kubernetes server-side-apply field-ownership conflicts — SSA ownership
 * is a separate concern handled by the apply mechanism.
 */
public enum WriteStrategy {
    UPSERT,
    PUT_IF_ABSENT
}

/**
 * Integration tests for the REAL {@link ExecutorRecoveryScheduler}. These use bounded waits
 * (latches with timeouts) to observe that a task actually ran — never sleeps as correctness
 * evidence. They verify scheduler mechanics and its wiring to {@link WatchLifecycle}.
 */
class ExecutorRecoverySchedulerIT {

    @Test @Timeout(10)
    void task_runs_on_executor_thread_not_inline() throws Exception {
        try (ExecutorRecoveryScheduler s = ExecutorRecoveryScheduler.createOwning("rec-test")) {
            Thread caller = Thread.currentThread();
            AtomicBoolean ranInline = new AtomicBoolean(false);
            CountDownLatch ran = new CountDownLatch(1);
            s.schedule(Duration.ofMillis(1), () -> {
                if (Thread.currentThread() == caller) ranInline.set(true);
                ran.countDown();
            });
            assertTrue(ran.await(5, TimeUnit.SECONDS), "task should run");
            assertFalse(ranInline.get(), "task must NOT run on the caller thread");
        }
    }

    @Test @Timeout(10)
    void cancel_prevents_a_not_yet_started_task() throws Exception {
        // Occupy the ONLY worker with a blocking task so nothing else can start; schedule the
        // target (0 delay) and CANCEL it while the worker is still blocked; queue a marker (0
        // delay) AFTER the target; release the worker; await the marker and assert the target
        // never ran. With cancel() a no-op the target would run before the marker -> test fails.
        ScheduledExecutorService exec = Executors.newSingleThreadScheduledExecutor();
        CountDownLatch workerBusy = new CountDownLatch(1);
        CountDownLatch releaseWorker = new CountDownLatch(1);
        CountDownLatch blockerDone = new CountDownLatch(1);
        java.util.concurrent.Future<?> blocker = null;
        try {
            ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(exec);
            // block the single worker; retain its Future so we can observe completion and surface
            // any failure, and ALWAYS release it in the finally block.
            blocker = exec.submit(() -> {
                try {
                    workerBusy.countDown();
                    if (!releaseWorker.await(5, TimeUnit.SECONDS)) throw new AssertionError("worker not released");
                } finally {
                    blockerDone.countDown();
                }
                return null;
            });
            assertTrue(workerBusy.await(5, TimeUnit.SECONDS), "worker should be occupied");

            AtomicInteger targetRuns = new AtomicInteger();
            var handle = s.schedule(Duration.ofMillis(0), targetRuns::incrementAndGet);
            handle.cancel();                               // cancel while worker is blocked
            CountDownLatch markerRan = new CountDownLatch(1);
            s.schedule(Duration.ofMillis(0), markerRan::countDown);  // queued after target

            releaseWorker.countDown();                     // now the queue drains in order
            assertTrue(markerRan.await(5, TimeUnit.SECONDS), "marker must run");
            assertEquals(0, targetRuns.get(), "cancelled target must NOT have run");
            assertTrue(blockerDone.await(5, TimeUnit.SECONDS), "the blocking task must complete");
            blocker.get(5, TimeUnit.SECONDS);              // surface any failure from the blocking task
        } finally {
            releaseWorker.countDown();                     // ensure the worker is never left parked
            exec.shutdownNow();
            assertTrue(exec.awaitTermination(5, TimeUnit.SECONDS), "executor must terminate");
        }
    }

    @Test @Timeout(10)
    void owning_scheduler_shuts_down_its_own_executor_on_close() throws Exception {
        // createOwning() OWNS its executor; close() must shut that executor down (and then reject
        // further scheduling). Contrast with the shared-executor test above, where close() is a
        // no-op. try-with-resources guarantees cleanup even if an assertion fails before the
        // explicit close; a second close() is a harmless no-op (executor already shut down).
        try (ExecutorRecoveryScheduler s = ExecutorRecoveryScheduler.createOwning("owned-test")) {
            CountDownLatch ran = new CountDownLatch(1);
            s.schedule(Duration.ofMillis(1), ran::countDown);
            assertTrue(ran.await(5, TimeUnit.SECONDS), "owned scheduler runs tasks before close");

            s.close();                                    // must shut down the OWNED executor

            // After close, scheduling must be rejected because the owned executor is shut down.
            assertThrows(RejectedExecutionException.class,
                    () -> s.schedule(Duration.ofMillis(1), () -> {}),
                    "an owned scheduler must reject scheduling after close() shuts its executor down");
        }
    }

    @Test @Timeout(10)
    void negative_delay_is_rejected() {
        try (ExecutorRecoveryScheduler s = ExecutorRecoveryScheduler.createOwning("rec-test")) {
            assertThrows(IllegalArgumentException.class,
                    () -> s.schedule(Duration.ofMillis(-1), () -> {}));
        }
    }

    @Test @Timeout(10)
    void shared_executor_is_not_shut_down_by_a_lifecycle_scheduler() throws Exception {
        ScheduledExecutorService shared = Executors.newSingleThreadScheduledExecutor();
        try {
            ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(shared);  // not owned
            s.close();                                   // must be a no-op for a shared executor
            assertFalse(shared.isShutdown(), "a shared executor must survive scheduler.close()");
            // and it still works:
            CountDownLatch ran = new CountDownLatch(1);
            s.schedule(Duration.ofMillis(1), ran::countDown);
            assertTrue(ran.await(5, TimeUnit.SECONDS));
        } finally {
            shared.shutdownNow();
        }
    }

    @Test @Timeout(10)
    void executor_rejection_surfaces_as_DEGRADED_in_the_lifecycle() throws Exception {
        // Wire a lifecycle to a scheduler whose executor is already shut down -> schedule() throws
        // RejectedExecutionException -> lifecycle must enter DEGRADED (agreed failure path).
        ScheduledExecutorService dead = Executors.newSingleThreadScheduledExecutor();
        dead.shutdownNow();
        ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(dead);

        final Runnable[] onClosedHolder = new Runnable[1];
        WatchFactory factory = onClosed -> {
            onClosedHolder[0] = onClosed;
            return (Watch) () -> onClosed.run();         // close() triggers onClosed
        };
        WatchLifecycle lc = new WatchLifecycle("deg", factory, s);
        lc.registerWatch();                              // installs a watch
        // trigger a close -> onClosed -> schedule recovery -> rejected -> DEGRADED
        onClosedHolder[0].run();
        assertEquals(WatchLifecycle.Health.DEGRADED, lc.health(),
                "rejected recovery scheduling must surface DEGRADED");
        assertFalse(lc.debugHasPendingRecovery(), "no phantom pending recovery after rejection");
    }

package io.ssaownership.reconciler;

import org.junit.jupiter.api.Test;

import java.time.Duration;
import java.util.concurrent.Delayed;
import java.util.concurrent.ScheduledFuture;
import java.util.concurrent.ScheduledThreadPoolExecutor;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.AtomicReference;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Direct unit tests for {@link ExecutorRecoveryScheduler}'s duration-conversion behaviour. These
 * assert the EXACT value submitted to the backing executor via a RECORDING executor, rather than
 * measuring elapsed time — so sub-millisecond rounding and overflow handling are observed
 * precisely and deterministically.
 */
class ExecutorRecoverySchedulerTest {

    /** Records the (delay, unit) of the last schedule() call; never actually runs the task. */
    static final class RecordingExecutor extends ScheduledThreadPoolExecutor {
        final AtomicLong lastDelay = new AtomicLong(Long.MIN_VALUE);
        final AtomicReference<TimeUnit> lastUnit = new AtomicReference<>();
        RecordingExecutor() { super(1); }
        @Override public ScheduledFuture<?> schedule(Runnable command, long delay, TimeUnit unit) {
            lastDelay.set(delay);
            lastUnit.set(unit);
            return new NoopFuture();
        }
    }

    static final class NoopFuture implements ScheduledFuture<Object> {
        @Override public long getDelay(TimeUnit unit) { return 0; }
        @Override public int compareTo(Delayed o) { return 0; }
        @Override public boolean cancel(boolean mayInterruptIfRunning) { return true; }
        @Override public boolean isCancelled() { return false; }
        @Override public boolean isDone() { return false; }
        @Override public Object get() { return null; }
        @Override public Object get(long timeout, TimeUnit unit) { return null; }
    }

    @Test @Timeout(10)
    void positive_sub_millisecond_delay_is_submitted_as_one_millisecond_not_zero() {
        RecordingExecutor rec = new RecordingExecutor();
        try {
            ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(rec);
            s.schedule(Duration.ofNanos(1), () -> {});            // positive, sub-millisecond
            assertEquals(TimeUnit.MILLISECONDS, rec.lastUnit.get());
            assertEquals(1L, rec.lastDelay.get(),
                    "a positive sub-ms delay must be submitted as 1ms, never silently 0");
        } finally {
            rec.shutdownNow();
        }
    }

    @Test @Timeout(10)
    void exactly_zero_delay_is_submitted_as_zero() {
        RecordingExecutor rec = new RecordingExecutor();
        try {
            ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(rec);
            s.schedule(Duration.ZERO, () -> {});
            assertEquals(TimeUnit.MILLISECONDS, rec.lastUnit.get(), "unit must be milliseconds");
            assertEquals(0L, rec.lastDelay.get(), "zero stays zero (not bumped to 1ms)");
        } finally {
            rec.shutdownNow();
        }
    }

    @Test @Timeout(10)
    void above_one_millisecond_truncates_fractional_millis_not_rounds_up() {
        RecordingExecutor rec = new RecordingExecutor();
        try {
            ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(rec);
            s.schedule(Duration.ofNanos(1_900_000), () -> {});    // 1.9 ms
            assertEquals(TimeUnit.MILLISECONDS, rec.lastUnit.get(), "unit must be milliseconds");
            assertEquals(1L, rec.lastDelay.get(), "1.9ms truncates to 1ms (toMillis), not up to 2ms");
        } finally {
            rec.shutdownNow();
        }
    }

    @Test @Timeout(10)
    void conversion_overflow_is_rejected_as_illegal_argument() {
        RecordingExecutor rec = new RecordingExecutor();
        try {
            ExecutorRecoveryScheduler s = new ExecutorRecoveryScheduler(rec);
            // A duration whose toMillis() overflows a long:
            Duration huge = Duration.ofSeconds(Long.MAX_VALUE);
            IllegalArgumentException ex = assertThrows(IllegalArgumentException.class,
                    () -> s.schedule(huge, () -> {}));
            assertTrue(ex.getMessage().toLowerCase().contains("too large"),
                    "overflow must be an IllegalArgumentException about representability, was: " + ex.getMessage());
            assertEquals(Long.MIN_VALUE, rec.lastDelay.get(),
                    "an overflowing delay must never reach the executor");
        } finally {
            rec.shutdownNow();
        }
    }
}

/**
 * Single-threaded in-memory {@link ConfigMapStore} fake for unit tests. It honours the policy
 * parts of the contract (missing-CM-throws-on-write, reject-null, preserve-unrelated, absence vs
 * empty) so a test cannot pass against behaviour the real adapter would not reproduce.
 *
 * NOT thread-safe and NOT evidence of concurrent putIfAbsent correctness — the deterministic
 * conditional-update race must be verified against a real API server (KubeConfigMapStoreIT), not
 * here. This fake only honours the single-threaded policy parts of the contract.
 */
final class InMemoryStore implements ConfigMapStore {

    private final Map<String, Map<String, String>> cms = new HashMap<>();
    int upsertCalls = 0;
    int putIfAbsentCalls = 0;

    /** When >0, the next N store calls of the matching op throw StoreException (retry tests). */
    int failNextGet = 0;
    int failNextUpsert = 0;
    int failNextPutIfAbsent = 0;

    /** Create an (empty) ConfigMap object, as the chart would. Writes require this to exist. */
    void createConfigMap(String cm) {
        cms.computeIfAbsent(cm, k -> new HashMap<>());
    }

    void seedKey(String cm, String key, String value) {
        Objects.requireNonNull(value, "value");   // reject null: a present key cannot have null value
        cms.computeIfAbsent(cm, k -> new HashMap<>()).put(key, value);
    }

    /** Delete a key but keep the ConfigMap object (models an external deletion of just the key). */
    void deleteKey(String cm, String key) {
        Map<String, String> data = cms.get(cm);
        if (data != null) {
            data.remove(key);
        }
    }

    @Override
    public Optional<String> get(String cm, String key) {
        if (failNextGet > 0) { failNextGet--; throw new StoreException("injected get failure"); }
        Map<String, String> data = cms.get(cm);
        if (data == null) {
            return Optional.empty();          // missing ConfigMap -> empty (per contract)
        }
        return Optional.ofNullable(data.get(key));  // empty ONLY if key absent; "" is present
    }

    @Override
    public void upsert(String cm, String key, String value) {
        Objects.requireNonNull(value, "value");   // reject null (per contract)
        if (failNextUpsert > 0) { failNextUpsert--; throw new StoreException("injected upsert failure"); }
        upsertCalls++;
        Map<String, String> data = cms.get(cm);
        if (data == null) {
            throw new StoreException("ConfigMap " + cm + " does not exist");  // do not create
        }
        data.put(key, value);                      // preserves unrelated keys
    }

    @Override
    public boolean putIfAbsent(String cm, String key, String value) {
        Objects.requireNonNull(value, "value");
        if (failNextPutIfAbsent > 0) { failNextPutIfAbsent--; throw new StoreException("injected putIfAbsent failure"); }
        putIfAbsentCalls++;
        Map<String, String> data = cms.get(cm);
        if (data == null) {
            throw new StoreException("ConfigMap " + cm + " does not exist");
        }
        if (data.containsKey(key)) {
            return false;                          // preserve existing
        }
        data.put(key, value);
        return true;
    }
}

package io.ssaownership.reconciler;

import java.time.Duration;
import java.util.ArrayList;
import java.util.List;

/**
 * Manually-controlled {@link RecoveryScheduler} for deterministic tests. It does NOT run tasks on
 * a timer — the test decides exactly when (and whether) a scheduled task runs, and can even run a
 * CANCELLED task on purpose to prove the lifecycle's own eligibility check (not cancellation) is
 * what protects against stale work. No clock, no sleeps.
 *
 * <h2>Threading restriction</h2>
 * This fake performs NO internal synchronisation of its bookkeeping ({@code scheduled},
 * {@code run}, {@code cancelled}). Two usage patterns are permitted:
 * <ul>
 *   <li>Single-threaded driving: the test thread calls {@code schedule} and the {@code run*}
 *       methods.</li>
 *   <li>Ordered cross-thread handoff: a {@code run*} method may be invoked from a different thread
 *       (e.g. the authorization-gap test runs a recovery task on a worker) PROVIDED the test
 *       establishes a happens-before edge — via latches/Futures — so that no two threads touch the
 *       bookkeeping concurrently. Each field is written by one thread and read by another only
 *       across such an edge.</li>
 * </ul>
 * UNSYNCHRONISED CONCURRENT access to the bookkeeping is prohibited; tests that need a genuinely
 * concurrent scheduler (handle-publication / rejection orderings) use {@code ControllableScheduler}
 * instead, which synchronises its state. In all cases a task BODY is executed WITHOUT holding any
 * lock, so a task that re-enters the lifecycle cannot deadlock against the scheduler.
 */
final class ManualScheduler implements RecoveryScheduler {

    static final class Task implements Cancellable {
        final Duration delay;
        final Runnable body;
        boolean cancelled = false;
        boolean run = false;
        Task(Duration delay, Runnable body) { this.delay = delay; this.body = body; }
        @Override public void cancel() { cancelled = true; }
    }

    final List<Task> scheduled = new ArrayList<>();
    boolean rejectScheduling = false;        // when true, schedule() throws (reject path test)

    @Override
    public Cancellable schedule(Duration delay, Runnable task) {
        if (rejectScheduling) {
            throw new RejectedException("scheduler is rejecting");
        }
        Task t = new Task(delay, task);
        scheduled.add(t);
        return t;
    }

    static final class RejectedException extends RuntimeException {
        RejectedException(String m) { super(m); }
    }

    /** Run the most recently scheduled task that is NOT yet run, regardless of cancellation. */
    void runLast() {
        Task t = null;
        for (int i = scheduled.size() - 1; i >= 0; i--) {
            if (!scheduled.get(i).run) { t = scheduled.get(i); break; }
        }
        if (t == null) throw new AssertionError("no not-yet-run task to run");
        t.run = true;
        t.body.run();                         // body runs with no lock held
    }

    /** Run a specific scheduled task even if cancelled — to prove stale-task protection. */
    void runEvenIfCancelled(Task t) { t.run = true; t.body.run(); }

    /** Tasks that are neither cancelled nor already run — i.e. genuinely still pending. */
    int pendingCount() {
        return (int) scheduled.stream().filter(t -> !t.cancelled && !t.run).count();
    }
    Task last() { return scheduled.get(scheduled.size() - 1); }
}

        }
        @Override public boolean putIfAbsent(String cm, String k, String v) {
            boolean wrote = delegate.putIfAbsent(cm, k, v);  // COMMIT
            if (armed) { armed = false; throw new StoreException("committed insert, then connection dropped"); }

/**
 * Deterministic tests for {@link WatchLifecycle}. A manually-controlled {@link ManualScheduler}
 * makes recovery timing explicit (no clock, no sleeps). Threaded tests use latches with CHECKED
 * waits, release gates in finally, and propagate worker exceptions through retained Futures, so a
 * green test cannot silently have skipped its intended interleaving. Timeouts are deadlock guards.
 */
class WatchLifecycleTest {

    /** A watch with identity that fires its bound onClosed exactly once on close. */
    static final class TestWatch implements Watch {
        final int id;
        final Runnable onClosed;
        final AtomicInteger closes = new AtomicInteger();
        TestWatch(int id, Runnable onClosed) { this.id = id; this.onClosed = onClosed; }
        @Override public void close() { if (closes.incrementAndGet() == 1) onClosed.run(); }
        boolean closed() { return closes.get() > 0; }
        @Override public String toString() { return "w#" + id; }
    }

    /** Factory recording every watch it makes. Subclassable for interleaving seams. */
    static class Factory implements WatchFactory {
        final Set<TestWatch> created = ConcurrentHashMap.newKeySet();
        final AtomicInteger seq = new AtomicInteger();
        volatile boolean closeImmediately = false;
        @Override public Watch create(Runnable onClosed) {
            TestWatch w = new TestWatch(seq.incrementAndGet(), onClosed);
            created.add(w);
            if (closeImmediately) w.close();
            return w;
        }
    }

    private TestWatch installedWatch(Factory f, WatchLifecycle lc) {
        return f.created.stream().filter(lc::debugIsInstalled).findFirst().orElseThrow();
    }
    private void assertNoLeak(Factory f, WatchLifecycle lc) {
        for (TestWatch w : f.created) {
            boolean inst = lc.debugIsInstalled(w);
            assertTrue(inst || w.closed(), w + " leaked (neither installed nor closed)");
            if (inst) assertFalse(w.closed(), w + " installed yet closed");
        }
    }

    // ---- basic lifecycle --------------------------------------------------------------------

    @Test @Timeout(10)
    void install_then_deactivate_closes_installed() {

        WatchLifecycle lc = new WatchLifecycle("t", f, s);
        lc.registerWatch();
        TestWatch w = installedWatch(f, lc);
        lc.deactivate();

        assertTrue(w.closed(), "installed watch closed on deactivate");
        assertNoLeak(f, lc);
    }

        Factory f = new Factory(); ManualScheduler s = new ManualScheduler();
        WatchLifecycle lc = new WatchLifecycle("t", f, s);
        lc.registerWatch();
        installedWatch(f, lc).close();                 // schedules recovery
        ManualScheduler.Task recovery = s.last();
        assertFalse(recovery.cancelled, "precondition: recovery not yet cancelled");

        assertTrue(recovery.cancelled, "deactivate must cancel the pending recovery handle");
    }

    // ---- scheduler handle-publication & shutdown/supersession orderings ---------------------
    //
    // These exercise the races the lifecycle documentation explicitly claims: the handle returned
    // by schedule() can arrive AFTER the task it schedules has already run (or after the lifecycle
    // was superseded/shut down), and a scheduling REJECTION can arrive after supersession/shutdown.
    // A controllable scheduler lets the test fix the ordering. Crucially, tasks run on a REAL
    // separate thread (never inline inside schedule()), so the harness does not itself violate the
    // "never inline" scheduler contract while verifying it.

    /**
     * Controllable {@link RecoveryScheduler}: captures each scheduled task; the test runs it (on a
     * dedicated thread) and independently decides when schedule() returns its handle, so handle
     * publication can be ordered before/after task execution, supersession, or shutdown. Bookkeeping
     * is synchronized; task bodies are NEVER invoked while holding the lock.
     */
    static final class ControllableScheduler implements RecoveryScheduler {
        static final class Handle implements Cancellable {
            final AtomicInteger cancels = new AtomicInteger();
            @Override public void cancel() { cancels.incrementAndGet(); }
            boolean cancelled() { return cancels.get() > 0; }
        }
        final Object mon = new Object();
        Runnable lastTask;                       // guarded by mon
        Handle lastHandle;                       // guarded by mon
        private CountDownLatch captured = new CountDownLatch(1);  // counted down when a task is captured
        private CountDownLatch releaseSchedule;  // if set, schedule() blocks until counted down
        private boolean rejectOnRelease = false; // if set, the released schedule() throws instead of returning

        /** Arrange that the NEXT schedule() call blocks (after capturing the task) until released. */
        CountDownLatch blockNextScheduleReturn() {
            CountDownLatch l = new CountDownLatch(1);
            synchronized (mon) { releaseSchedule = l; rejectOnRelease = false; }
            return l;
        }
        /** Like {@link #blockNextScheduleReturn} but the released schedule() THROWS a rejection. */
        CountDownLatch blockThenRejectNextSchedule() {
            CountDownLatch l = new CountDownLatch(1);
            synchronized (mon) { releaseSchedule = l; rejectOnRelease = true; }
            return l;
        }

        /** Block until a task has been handed to schedule() (replaces test spin loops). */
        void awaitCaptured(long seconds) {
            CountDownLatch c;
            synchronized (mon) { c = captured; }
            try { if (!c.await(seconds, TimeUnit.SECONDS)) throw new AssertionError("no task captured in time"); }
            catch (InterruptedException e) { Thread.currentThread().interrupt(); throw new AssertionError(e); }
        }

        @Override public Cancellable schedule(Duration delay, Runnable task) {
            CountDownLatch toAwait;
            boolean reject;
            Handle h = new Handle();
            synchronized (mon) {
                lastTask = task; lastHandle = h;
                toAwait = releaseSchedule; releaseSchedule = null;
                reject = rejectOnRelease; rejectOnRelease = false;
                captured.countDown();            // signal capture AFTER recording task+handle
            }
            if (toAwait != null) {
                try { if (!toAwait.await(2, TimeUnit.SECONDS)) throw new AssertionError("schedule() not released"); }
                catch (InterruptedException e) { Thread.currentThread().interrupt(); throw new AssertionError(e); }
            }
            if (reject) throw new ManualScheduler.RejectedException("rejected on release");
            return h;
        }
        Runnable task()  { synchronized (mon) { return lastTask; } }
        Handle handle()  { synchronized (mon) { return lastHandle; } }
    }

    private Future<?> runOnThread(ExecutorService pool, Runnable r) {
        return pool.submit(() -> { r.run(); return null; });
    }

    @Test @Timeout(10)
    void recovery_runs_before_schedule_returns_its_handle_then_handle_is_cancelled() throws Exception {
        // Ordering: schedule() captures the task but BLOCKS before returning its handle. The task
        // runs to completion on another thread (consuming the reservation, installing B). Only then
        // does schedule() return. The lifecycle must observe the reservation is no longer pending
        // and CANCEL the late handle rather than record it.
        ControllableScheduler s = new ControllableScheduler();
        Factory f = new Factory();
        WatchLifecycle lc = new WatchLifecycle("t", f, s);
        ExecutorService pool = Executors.newFixedThreadPool(2);
        try {
            CountDownLatch releaseSchedule = s.blockNextScheduleReturn();
            lc.registerWatch();                              // A
            // closing A calls scheduleRecovery, which will block inside schedule():
            Future<?> closeFut = runOnThread(pool, () -> installedWatch(f, lc).close());
            // wait until the task was captured (schedule() entered):
            s.awaitCaptured(2);

            Future<?> runFut = runOnThread(pool, s.task());
            runFut.get(5, TimeUnit.SECONDS);                 // recovery installs B, consumes reservation

            assertEquals(1, f.created.stream().filter(lc::debugIsInstalled).count(), "B installed");
            assertFalse(lc.debugHasPendingRecovery());

        }
    }

        } finally {

            assertTrue(pool.awaitTermination(5, TimeUnit.SECONDS));
        }
        assertEquals(1, f.created.stream().filter(lc::debugIsInstalled).count());
        assertNoLeak(f, lc);
    }
}

package io.ssaownership.schemasync;

import io.fabric8.kubernetes.client.KubernetesClient;
import io.fabric8.kubernetes.client.dsl.base.PatchContext;
import io.fabric8.kubernetes.client.dsl.base.PatchType;
import io.ssaownership.schemasync.model.SchemaBinding;
import io.ssaownership.schemasync.model.SchemaBindingSpec;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * BROKEN SchemaSync operator.
 *
 * <p>On each reconcile it performs a server-side apply as field manager
 * {@code schemasync-operator} that INCLUDES {@code spec.schemaVersion} — a field the deployment
 * (Helm chart) also sets. Under SSA that makes the operator a co-owner of a desired-state field
 * it has no business owning. A subsequent chart apply (different manager) then CONFLICTS on
 * {@code spec.schemaVersion}.
 *
 * <p>This is the generic reproduction of the real-world defect: a controller contending for a
 * field whose desired state belongs to another actor. The fix is {@link FixedSchemaSyncReconciler}.
 */
public final class BrokenSchemaSyncReconciler {

    private static final Logger LOG = LoggerFactory.getLogger(BrokenSchemaSyncReconciler.class);
    public static final String FIELD_MANAGER = "schemasync-operator";

    private final KubernetesClient client;
    private final String namespace;

    public BrokenSchemaSyncReconciler(KubernetesClient client, String namespace) {
        this.client = java.util.Objects.requireNonNull(client, "client");
        this.namespace = java.util.Objects.requireNonNull(namespace, "namespace");
    }

    /** Reconcile: (wrongly) server-side-apply the spec, co-owning spec.schemaVersion. */
    public void reconcile(String name) {
        SchemaBinding current = client.resources(SchemaBinding.class)
                .inNamespace(namespace).withName(name).get();
        if (current == null) return;
        String observed = current.getSpec() != null ? current.getSpec().getSchemaVersion() : null;

        // Build an apply patch that INCLUDES spec.schemaVersion (the bug): the operator claims
        // ownership of the chart's desired-state field.
        SchemaBinding patch = new SchemaBinding();
        patch.getMetadata().setName(name);
        SchemaBindingSpec spec = new SchemaBindingSpec();
        spec.setSchemaVersion(observed);     // echoes/claims the chart-owned field under SSA
        patch.setSpec(spec);

        client.resources(SchemaBinding.class).inNamespace(namespace).withName(name)
                .patch(new PatchContext.Builder()
                                .withPatchType(PatchType.SERVER_SIDE_APPLY)
                                .withFieldManager(FIELD_MANAGER)
                                .withForce(false)
                                .build(),
                        patch);
        LOG.info("[BROKEN] applied spec (co-owning spec.schemaVersion) for {}", name);
    }
}

package io.ssaownership.schemasync;

import io.fabric8.kubernetes.client.KubernetesClient;
import io.fabric8.kubernetes.client.dsl.base.PatchContext;
import io.fabric8.kubernetes.client.dsl.base.PatchType;
import io.ssaownership.schemasync.model.SchemaBinding;
import io.ssaownership.schemasync.model.SchemaBindingStatus;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * FIXED SchemaSync operator.
 *
 * <p>It records OBSERVED state in {@code status} (via the /status subresource) and NEVER writes
 * {@code spec}. The decisive architectural property is WHERE it writes: {@code status}, not
 * {@code spec}. Because it never applies {@code spec}, it never contends with the chart's
 * ownership of {@code spec.schemaVersion}, so a chart apply never conflicts with this operator.
 *
 * <p><b>Update semantics (precise).</b> {@link #reconcile} uses {@code updateStatus()} — a plain
 * update of the {@code /status} subresource (optimistic-concurrency on resourceVersion). It is
 * NOT a server-side apply and does NOT register {@link #FIELD_MANAGER} as an SSA owner of status.
 * This class therefore does not claim SSA ownership of {@code status}; it claims only that it
 * leaves {@code spec} untouched. (If strict SSA status ownership under a named manager were
 * required, {@code reconcile} would instead do a server-side apply of {@code status} with
 * {@code FIELD_MANAGER}; that is a deliberate non-goal here.)
 *
 * <h2>Ownership handoff (upgrade from the broken operator)</h2>
 * Simply ceasing to write {@code spec} does NOT relinquish an EXISTING server-side-apply field
 * ownership that the broken operator established on {@code spec.schemaVersion}. SSA ownership
 * persists until explicitly released. {@link #releaseStaleSpecOwnership} performs the handoff:
 * it re-applies as the SAME field manager with a patch that OWNS NOTHING in spec, which drops the
 * manager's claim on {@code spec.schemaVersion} WITHOUT deleting the field's value — <b>but only
 * because another manager (the chart) still owns that field's value</b>. See the precondition on
 * {@link #releaseStaleSpecOwnership}.
 */
public final class FixedSchemaSyncReconciler {

    private static final Logger LOG = LoggerFactory.getLogger(FixedSchemaSyncReconciler.class);
    public static final String FIELD_MANAGER = "schemasync-operator";

    private final KubernetesClient client;
    private final String namespace;

    public FixedSchemaSyncReconciler(KubernetesClient client, String namespace) {
        this.client = java.util.Objects.requireNonNull(client, "client");
        this.namespace = java.util.Objects.requireNonNull(namespace, "namespace");
    }

    /** Reconcile: write ONLY status.observedSchemaVersion (never spec). */
    public void reconcile(String name) {
        SchemaBinding current = client.resources(SchemaBinding.class)
                .inNamespace(namespace).withName(name).get();
        if (current == null) return;
        String desired = current.getSpec() != null ? current.getSpec().getSchemaVersion() : null;

        SchemaBindingStatus status = new SchemaBindingStatus();
        status.setObservedSchemaVersion(desired);
        status.setPhase(desired == null ? "Pending" : "Synced");
        current.setStatus(status);

        // updateStatus targets the /status subresource — spec is untouched, no spec ownership.
        client.resources(SchemaBinding.class).inNamespace(namespace).resource(current).updateStatus();
        LOG.info("[FIXED] wrote status.observedSchemaVersion={} for {} (spec untouched)", desired, name);
    }

    /**
     * Ownership handoff: drop this manager's SSA claim on {@code spec} established by the broken
     * operator, WITHOUT deleting spec.schemaVersion.
     *
     * <p><b>PRECONDITION (critical).</b> This is safe to use for migration ONLY when another
     * manager (here, the chart) also owns {@code spec.schemaVersion}. Re-applying as this manager
     * with an empty spec removes the fields THIS manager owned in spec; a field co-owned by the
     * chart keeps the chart's value. If this operator were the SOLE owner of the field, dropping
     * its claim could remove the value (or fail validation on a required field). Do not use this as
     * a blind production migration without first confirming another owner exists — the SSA
     * demonstration test asserts the value is preserved precisely because the chart co-owns it.
     */
    public void releaseStaleSpecOwnership(String name) {
        SchemaBinding emptyApply = new SchemaBinding();
        emptyApply.getMetadata().setName(name);
        // No spec set -> this manager, after this apply, owns no spec fields.
        client.resources(SchemaBinding.class).inNamespace(namespace).withName(name)
                .patch(new PatchContext.Builder()
                                .withPatchType(PatchType.SERVER_SIDE_APPLY)
                                .withFieldManager(FIELD_MANAGER)
                                .withForce(false)
                                .build(),
                        emptyApply);
        LOG.info("[FIXED] released stale spec ownership for {} (value preserved by the chart's manager)", name);
    }
}

package io.ssaownership.schemasync;

import io.fabric8.kubernetes.api.model.ManagedFieldsEntry;
import io.fabric8.kubernetes.api.model.NamespaceBuilder;
import io.fabric8.kubernetes.api.model.apiextensions.v1.CustomResourceDefinition;
import io.fabric8.kubernetes.client.KubernetesClient;

import io.fabric8.kubernetes.client.dsl.base.PatchContext;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Live SSA ownership demonstration on a real cluster, with ACTUAL managedFields inspection:
 *  (1) the broken operator co-owning spec.schemaVersion makes a chart apply conflict (HTTP 409);
 *  (2) the fixed operator (status-only) does NOT conflict and introduces no spec claim;
 *  (3) the ownership handoff drops the broken operator's spec claim while the chart's value
 *      survives — and the test asserts the PRECONDITION (the chart co-owns the field) holds.
 * Named field managers, real 409 codes, and managedFields assertions make this diagnostic.
 *
 * Gated by a live-integration profile (see failsafe config). {@link #clusterAvailable()} only
 * decides local enablement; in CI the profile makes an unreachable cluster a FAILURE, not a skip.
 */
@EnabledIf("clusterAvailable")
class SchemaSyncOwnershipIT {

    static final String CHART_MANAGER = "chart-ssa-manager";   // models a chart/deployment SSA writer
    static boolean clusterAvailable() {
        if ("true".equalsIgnoreCase(System.getProperty("ssa.it.requireCluster"))) {
            return true;   // profile active: do NOT skip — let a real failure surface
        }
        try (KubernetesClient c = new KubernetesClientBuilder().build()) { c.namespaces().list(); return true; }
        catch (Exception e) { return false; }
    }

    private KubernetesClient client;
    private String ns;
    private final String name = "demo-binding";

    @BeforeEach
    void setUp() throws Exception {
        client = new KubernetesClientBuilder().build();
        ns = "ssa-op-it-" + UUID.randomUUID().toString().substring(0, 8);
        client.namespaces().resource(new NamespaceBuilder()
                .withNewMetadata().withName(ns).endMetadata().build()).create();
        try (InputStream crd = getClass().getResourceAsStream("/crd/schemabinding-crd.yaml")) {
            Objects.requireNonNull(crd, "missing /crd/schemabinding-crd.yaml on the test classpath");
            CustomResourceDefinition def = client.apiextensions().v1()
                    .customResourceDefinitions().load(crd).item();
            // The /status subresource MUST be configured for updateStatus() to target it.
            boolean statusEnabled = def.getSpec().getVersions().stream()
                    .anyMatch(v -> v.getSubresources() != null && v.getSubresources().getStatus() != null);
            assertTrue(statusEnabled, "CRD must enable the status subresource");
            client.apiextensions().v1().customResourceDefinitions().resource(def).serverSideApply();
        }
        awaitCrdEstablished("schemabindings.ssaownership.io", Duration_SECONDS(30));
    }

    @AfterEach
    void tearDown() {
        // NOTE: the SchemaBinding CRD is CLUSTER-SCOPED and intentionally left installed (shared
        // fixture). Namespace cleanup does not remove it; run these on a disposable/dedicated
        // cluster. We delete only our namespace here.
        try { client.namespaces().withName(ns).delete(); } finally { client.close(); }
    }

    private static long Duration_SECONDS(long s) { return TimeUnit.SECONDS.toNanos(s); }

    /** Wait, with a bound, for the CRD's Established condition to be True. Restores interruption. */
    private void awaitCrdEstablished(String crdName, long budgetNanos) {
        long deadline = System.nanoTime() + budgetNanos;
        while (System.nanoTime() < deadline) {
            CustomResourceDefinition d = client.apiextensions().v1()
                    .customResourceDefinitions().withName(crdName).get();
            if (d != null && d.getStatus() != null && d.getStatus().getConditions() != null
                    && d.getStatus().getConditions().stream()
                        .anyMatch(c -> "Established".equals(c.getType()) && "True".equals(c.getStatus()))) {
                return;
            }
            try { TimeUnit.MILLISECONDS.sleep(100); }
            catch (InterruptedException e) { Thread.currentThread().interrupt(); throw new AssertionError("interrupted waiting for CRD", e); }
        }
        throw new AssertionError("CRD " + crdName + " did not reach Established=True within budget");
    }

    private void chartApply(String version, boolean force) {
        SchemaBinding sb = new SchemaBinding();
        sb.getMetadata().setName(name);
        SchemaBindingSpec spec = new SchemaBindingSpec();
        spec.setSchemaVersion(version);
        sb.setSpec(spec);
        client.resources(SchemaBinding.class).inNamespace(ns).withName(name)
                .patch(new PatchContext.Builder().withPatchType(PatchType.SERVER_SIDE_APPLY)
                        .withFieldManager(CHART_MANAGER).withForce(force).build(), sb);
    }

    private String specVersion() {
        SchemaBinding sb = client.resources(SchemaBinding.class).inNamespace(ns).withName(name).get();
        return sb != null && sb.getSpec() != null ? sb.getSpec().getSchemaVersion() : null;
    }
    private String statusObserved() {
        SchemaBinding sb = client.resources(SchemaBinding.class).inNamespace(ns).withName(name).get();
        return sb != null && sb.getStatus() != null ? sb.getStatus().getObservedSchemaVersion() : null;
    }

    /**
     * True iff {@code manager} owns {@code spec.schemaVersion} via an Apply operation, determined
     * by STRUCTURED traversal of the managedFields fieldsV1 tree (serialized to JSON), not by a
     * fragile toString() substring. The SSA fieldset encodes paths as nested keys {@code f:spec}
     * then {@code f:schemaVersion}.
     */
    private boolean ownsSpecSchemaVersion(String manager) {
        for (ManagedFieldsEntry e : managedFieldsFor(manager, "Apply")) {
            com.fasterxml.jackson.databind.JsonNode spec = fieldsTree(e).get("f:spec");
            if (spec != null && spec.has("f:schemaVersion")) return true;
        }
        return false;
    }

    /** True iff {@code manager} owns ANY path beneath {@code f:spec} (Apply operation). */
    private boolean ownsAnySpecPath(String manager) {
        for (ManagedFieldsEntry e : managedFieldsFor(manager, "Apply")) {
            com.fasterxml.jackson.databind.JsonNode spec = fieldsTree(e).get("f:spec");
            if (spec != null && spec.size() > 0) return true;
        }
        return false;
    }

    private List<ManagedFieldsEntry> managedFieldsFor(String manager, String operation) {
        SchemaBinding sb = client.resources(SchemaBinding.class).inNamespace(ns).withName(name).get();
        List<ManagedFieldsEntry> mf = sb.getMetadata().getManagedFields();
        if (mf == null) return List.of();
        return mf.stream()
                .filter(e -> manager.equals(e.getManager()))
                .filter(e -> operation == null || operation.equals(e.getOperation()))
                .toList();
    }

    /** Serialize an entry's fieldsV1 to a JSON tree for structured path inspection. */
    private com.fasterxml.jackson.databind.JsonNode fieldsTree(ManagedFieldsEntry e) {
        try {
            Object props = e.getFieldsV1() == null ? null : e.getFieldsV1().getAdditionalProperties();
            String json = io.fabric8.kubernetes.client.utils.Serialization.asJson(props == null ? java.util.Map.of() : props);
            return new com.fasterxml.jackson.databind.ObjectMapper().readTree(json);
        } catch (Exception ex) {
            throw new AssertionError("failed to parse managedFields fieldsV1", ex);
        }
    }

    @Test @Timeout(120)
    void broken_operator_causes_a_chart_apply_conflict_409_and_co_ownership_is_visible() {
        chartApply("1.0", false);
        assertTrue(ownsSpecSchemaVersion(CHART_MANAGER), "precondition: chart owns spec.schemaVersion");

        new BrokenSchemaSyncReconciler(client, ns).reconcile(name);
        assertTrue(ownsSpecSchemaVersion(BrokenSchemaSyncReconciler.FIELD_MANAGER),
                "broken operator must CO-OWN spec.schemaVersion (visible in managedFields)");

        KubernetesClientException ex = assertThrows(KubernetesClientException.class,
                () -> chartApply("2.0", false),
                "chart apply must conflict when the operator co-owns spec.schemaVersion");
        assertEquals(409, ex.getCode(), "the failure must be an SSA field-ownership conflict (HTTP 409)");
    }

    @Test @Timeout(120)
    void fixed_operator_writes_status_only_and_introduces_no_spec_claim() {
        chartApply("1.0", false);
        new FixedSchemaSyncReconciler(client, ns).reconcile(name);

        assertEquals("1.0", statusObserved(), "fixed operator records observed state in status");
        assertEquals("1.0", specVersion(), "spec untouched by the operator");
        // the operator must NOT own ANY path under spec (not merely spec.schemaVersion):
        assertFalse(ownsAnySpecPath(FixedSchemaSyncReconciler.FIELD_MANAGER),
                "status-only operator must not own any spec field");

        assertDoesNotThrow(() -> chartApply("2.0", false),
                "chart apply must NOT conflict when the operator owns no spec field");
        assertEquals("2.0", specVersion());
    }

    @Test @Timeout(120)
    void ownership_handoff_drops_broken_spec_claim_but_chart_value_survives() {
        // Broken state: operator co-owns spec.schemaVersion.
        chartApply("1.0", false);
        new BrokenSchemaSyncReconciler(client, ns).reconcile(name);
        assertTrue(ownsSpecSchemaVersion(BrokenSchemaSyncReconciler.FIELD_MANAGER), "operator co-owns spec");
        KubernetesClientException pre = assertThrows(KubernetesClientException.class, () -> chartApply("2.0", false));
        assertEquals(409, pre.getCode(), "precondition conflict must be a 409");

        // HANDOFF PRECONDITION (explicit): the chart STILL co-owns the field, so dropping the
        // operator's claim cannot delete the value.
        assertTrue(ownsSpecSchemaVersion(CHART_MANAGER),
                "handoff precondition: another manager (chart) owns spec.schemaVersion");

        FixedSchemaSyncReconciler fixed = new FixedSchemaSyncReconciler(client, ns);
        fixed.releaseStaleSpecOwnership(name);

        assertFalse(ownsSpecSchemaVersion(BrokenSchemaSyncReconciler.FIELD_MANAGER),
                "handoff must remove the operator's spec.schemaVersion claim");
        assertTrue(ownsSpecSchemaVersion(CHART_MANAGER), "chart still owns the field after handoff");
        assertEquals("1.0", specVersion(), "handoff must NOT delete the chart's spec value");

        assertDoesNotThrow(() -> chartApply("2.0", false), "after handoff the chart apply is conflict-free");
        assertEquals("2.0", specVersion());

        fixed.reconcile(name);
        assertEquals("2.0", statusObserved());
        assertFalse(ownsAnySpecPath(FixedSchemaSyncReconciler.FIELD_MANAGER),
                "the fixed operator still introduces no spec claim");
    }
}
